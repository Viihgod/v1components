type PongColors = {
  background: string;
  surface: string;
  accent: string;
  text: string;
  line: string;
};

type TrailPoint = {
  x: number;
  y: number;
};

function getPongColors(element: HTMLElement): PongColors {
  const styles = getComputedStyle(element);

  return {
    background: styles.getPropertyValue("--color-bg").trim() || "#010600",
    surface: styles.getPropertyValue("--color-surface-raised").trim() || "#0d170d",
    accent: styles.getPropertyValue("--color-clear-lime").trim() || "#c3f36b",
    text: styles.getPropertyValue("--color-text").trim() || "#f3f8f0",
    line: styles.getPropertyValue("--color-line").trim() || "#233a24",
  };
}

function drawRoundedRect(
  context: CanvasRenderingContext2D,
  x: number,
  y: number,
  width: number,
  height: number,
  radius: number,
) {
  context.beginPath();
  if (typeof context.roundRect === "function") {
    context.roundRect(x, y, width, height, radius);
  } else {
    context.moveTo(x + radius, y);
    context.arcTo(x + width, y, x + width, y + height, radius);
    context.arcTo(x + width, y + height, x, y + height, radius);
    context.arcTo(x, y + height, x, y, radius);
    context.arcTo(x, y, x + width, y, radius);
  }
  context.fill();
}

function drawPaddleFlame(
  context: CanvasRenderingContext2D,
  x: number,
  y: number,
  colors: PongColors,
  time: number,
  animate: boolean,
) {
  const flicker = animate ? Math.sin(time / 78) * 1.6 : 0;
  const tip = 16 + flicker;

  context.save();
  context.fillStyle = colors.accent;
  context.shadowColor = colors.accent;
  context.shadowBlur = 18;
  context.beginPath();
  context.moveTo(x - 1, y - PONG_PADDLE_HEIGHT / 2 + 4);
  context.quadraticCurveTo(x - 6, y - 13, x - tip, y - 11);
  context.quadraticCurveTo(x - tip + 4, y - 4, x - tip - 1, y);
  context.quadraticCurveTo(x - tip + 4, y + 4, x - tip, y + 11);
  context.quadraticCurveTo(x - 6, y + 13, x - 1, y + PONG_PADDLE_HEIGHT / 2 - 4);
  context.closePath();
  context.fill();

  context.fillStyle = colors.text;
  context.globalAlpha = 0.82;
  context.shadowBlur = 8;
  context.beginPath();
  context.moveTo(x - 2, y - 14);
  context.quadraticCurveTo(x - 6, y - 7, x - 10, y);
  context.quadraticCurveTo(x - 6, y + 7, x - 2, y + 14);
  context.closePath();
  context.fill();
  context.restore();
}

function drawPong(
  context: CanvasRenderingContext2D,
  state: PongState,
  width: number,
  height: number,
  trail: TrailPoint[],
  showTrail: boolean,
  flashSide: "player" | "cpu" | null,
  colors: PongColors,
  blastEnabled: boolean,
  blastArmed: boolean,
  time: number,
  animateFlame: boolean,
) {
  context.clearRect(0, 0, width, height);
  context.fillStyle = colors.background;
  context.fillRect(0, 0, width, height);

  context.strokeStyle = colors.line;
  context.lineWidth = 1;
  context.setLineDash([2, 8]);
  context.beginPath();
  context.moveTo(width / 2, 18);
  context.lineTo(width / 2, height - 18);
  context.stroke();
  context.setLineDash([]);

  if (showTrail) {
    trail.forEach((point, index) => {
      const opacity = ((index + 1) / trail.length) * 0.32;
      context.globalAlpha = opacity;
      context.fillStyle = colors.accent;
      context.beginPath();
      context.arc(point.x, point.y, 2 + (index / trail.length) * 1.5, 0, Math.PI * 2);
      context.fill();
    });
    context.globalAlpha = 1;
  }

  const playerX = 18;
  const cpuX = width - 18;
  const playerFlash = flashSide === "player";
  const cpuFlash = flashSide === "cpu";

  for (const paddle of [
    { x: playerX, y: state.playerY, flash: playerFlash },
    { x: cpuX, y: state.cpuY, flash: cpuFlash },
  ]) {
    if (paddle.x === playerX && blastEnabled && blastArmed) {
      drawPaddleFlame(context, paddle.x, paddle.y, colors, time, animateFlame);
    }

    context.save();
    context.fillStyle = paddle.flash ? colors.text : colors.accent;
    context.shadowColor = colors.accent;
    context.shadowBlur = paddle.flash ? 14 : 0;
    drawRoundedRect(
      context,
      paddle.x - PONG_PADDLE_WIDTH / 2,
      paddle.y - PONG_PADDLE_HEIGHT / 2,
      PONG_PADDLE_WIDTH,
      PONG_PADDLE_HEIGHT,
      3,
    );
    context.restore();
  }

  context.save();
  context.fillStyle = colors.accent;
  context.shadowColor = colors.accent;
  context.shadowBlur = 12;
  context.beginPath();
  context.arc(state.ball.x, state.ball.y, state.ball.radius, 0, Math.PI * 2);
  context.fill();
  context.restore();

  context.fillStyle = colors.surface;
  context.globalAlpha = 0.7;
  context.fillRect(12, height - 12, width - 24, 1);
  context.globalAlpha = 1;
}

function SettingsToggle({
  label,
  description,
  checked,
  onToggle,
}: {
  label: string;
  description: string;
  checked: boolean;
  onToggle: () => void;
}) {
  return (
    <div
      className="pong-game__toggle"
      role="switch"
      aria-checked={checked}
      data-checked={checked}
      tabIndex={0}
      onClick={onToggle}
      onKeyDown={(event) => {
        if (event.key === "Enter" || event.key === " ") {
          event.preventDefault();
          onToggle();
        }
      }}
    >
      <span className="pong-game__toggle-copy">
        <strong>{label}</strong>
        <small>{description}</small>
      </span>
      <span className="pong-game__toggle-control">
        <input
          className="pong-game__toggle-input"
          type="checkbox"
          checked={checked}
          onChange={onToggle}
          aria-hidden="true"
          tabIndex={-1}
          onClick={(event) => event.stopPropagation()}
        />
        <span className="pong-game__toggle-track" aria-hidden="true">
          <span className="pong-game__toggle-thumb" />
        </span>
      </span>
    </div>
  );
}

export type PongGameProps = ViVisualProps & {
  blast?: boolean;
  blastBars?: BlastBars;
  className?: string;
  onBlastBarsChange?: (value: BlastBars) => void;
};

export function PongGame({
  blast = false,
  blastBars = DEFAULT_BLAST_BARS,
  className = "",
  onBlastBarsChange,
  shinyBorder = DEFAULT_VI_VISUAL_CONFIG.shinyBorder,
  glass = DEFAULT_VI_VISUAL_CONFIG.glass,
  radius = "none",
  activeColor = DEFAULT_VI_VISUAL_CONFIG.activeColor,
  standbyColor = DEFAULT_VI_VISUAL_CONFIG.standbyColor,
  activeTextColor = DEFAULT_VI_VISUAL_CONFIG.activeTextColor,
  standbyTextColor = DEFAULT_VI_VISUAL_CONFIG.standbyTextColor,
  borderColor = DEFAULT_VI_VISUAL_CONFIG.borderColor,
  shineColorA = DEFAULT_VI_VISUAL_CONFIG.shineColorA,
  shineColorB = DEFAULT_VI_VISUAL_CONFIG.shineColorB,
}: PongGameProps) {
  const stageRef = useRef<HTMLDivElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const engineRef = useRef<PongState | null>(null);
  const dimensionsRef = useRef({ width: 360, height: 190, dpr: 1 });
  const playerTargetYRef = useRef(95);
  const settingsRef = useRef<PongSettings>(defaultSettings);
  const reducedMotionRef = useRef(false);
  const activeRef = useRef(false);
  const blastEnabledRef = useRef(blast);
  const blastBarsRef = useRef<BlastBars>(blastBars);
  const blastArmedRef = useRef(false);
  const blastChargeRef = useRef(0);
  const trailRef = useRef<TrailPoint[]>([]);
  const mobileControlDirectionRef = useRef<PongControlDirection | null>(null);
  const impactFlashRef = useRef<{ side: "player" | "cpu" | null; until: number }>({
    side: null,
    until: 0,
  });
  const shakeTimerRef = useRef<number | null>(null);
  const [settings, setSettings] = useState(defaultSettings);
  const [settingsOpen, setSettingsOpen] = useState(false);
  const [score, setScore] = useState<Score>({ player: 0, cpu: 0 });
  const [isEngaged, setIsEngaged] = useState(false);
  const [internalBlastBars, setInternalBlastBars] = useState<BlastBars>(blastBars);
  const [blastCharge, setBlastCharge] = useState(0);
  const [blastArmed, setBlastArmed] = useState(false);
  const reducedMotion = useReducedMotion();
  const selectedBlastBars = onBlastBarsChange ? blastBars : internalBlastBars;
  const visualConfig = createViVisualConfig({
    shinyBorder,
    glass,
    radius,
    activeColor,
    standbyColor,
    activeTextColor,
    standbyTextColor,
    borderColor,
    shineColorA,
    shineColorB,
  });

  const blastReady = blast && blastCharge >= selectedBlastBars && !blastArmed;

  useEffect(() => {
    settingsRef.current = settings;
    if (!settings.trail) {
      trailRef.current = [];
    }
  }, [settings]);

  useEffect(() => {
    reducedMotionRef.current = reducedMotion ?? false;
  }, [reducedMotion]);

  useEffect(() => {
    blastEnabledRef.current = blast;

    if (!blast) {
      blastArmedRef.current = false;
      blastChargeRef.current = 0;
      if (engineRef.current) {
        engineRef.current.blastCharge = 0;
      }
    }
  }, [blast]);

  useEffect(() => {
    blastBarsRef.current = selectedBlastBars;
    const nextCharge = Math.min(blastChargeRef.current, selectedBlastBars);
    blastChargeRef.current = nextCharge;

    if (engineRef.current) {
      engineRef.current.blastCharge = nextCharge;
    }

    setBlastCharge(nextCharge);
  }, [selectedBlastBars]);

  useEffect(() => {
    const canvasElement = canvasRef.current;
    const stageElement = stageRef.current;
    if (!canvasElement || !stageElement) {
      return;
    }

    const canvasNode = canvasElement;
    const stageNode = stageElement;
    const context = canvasNode.getContext("2d");
    if (!context) {
      return;
    }
    const drawingContext = context;

    const colors = getPongColors(stageNode);
    let animationFrame = 0;
    let previousTime = performance.now();

    function resizeCanvas() {
      const rect = stageNode.getBoundingClientRect();
      const width = Math.max(280, rect.width);
      const height = Math.max(170, rect.height);
      const dpr = Math.min(window.devicePixelRatio || 1, 2);

      dimensionsRef.current = { width, height, dpr };
      canvasNode.width = Math.round(width * dpr);
      canvasNode.height = Math.round(height * dpr);
      drawingContext.setTransform(dpr, 0, 0, dpr, 0, 0);

      if (!engineRef.current) {
        engineRef.current = createPongState(width, height);
        playerTargetYRef.current = height / 2;
      } else {
        engineRef.current.playerY = clamp(
          engineRef.current.playerY,
          PONG_PADDLE_HEIGHT / 2,
          height - PONG_PADDLE_HEIGHT / 2,
        );
        engineRef.current.cpuY = clamp(
          engineRef.current.cpuY,
          PONG_PADDLE_HEIGHT / 2,
          height - PONG_PADDLE_HEIGHT / 2,
        );
      }
    }

    function triggerFeedback(event: PongStepEvent, time: number) {
      const currentSettings = settingsRef.current;
      if (currentSettings.impactFlash && (event.playerHit || event.cpuHit)) {
        impactFlashRef.current = {
          side: event.playerHit ? "player" : "cpu",
          until: time + 100,
        };
      }

      if (
        currentSettings.screenShake &&
        !reducedMotionRef.current &&
        (event.point || event.highSpeedHit)
      ) {
        stageNode.classList.remove("pong-game__stage--shake");
        void stageNode.offsetWidth;
        stageNode.classList.add("pong-game__stage--shake");
        if (shakeTimerRef.current !== null) {
          window.clearTimeout(shakeTimerRef.current);
        }
        shakeTimerRef.current = window.setTimeout(() => {
          stageNode.classList.remove("pong-game__stage--shake");
          shakeTimerRef.current = null;
        }, 280);
      }
    }

    function render(time: number) {
      const deltaSeconds = Math.min((time - previousTime) / 1000, 0.032);
      previousTime = time;
      const state = engineRef.current;

      if (state) {
        const currentSettings = settingsRef.current;
        let event: PongStepEvent = {
          playerHit: false,
          cpuHit: false,
          point: null,
          highSpeedHit: false,
          blastHit: false,
          blastCharge: state.blastCharge,
        };

        if (activeRef.current) {
          const mobileControlDirection = mobileControlDirectionRef.current;
          if (mobileControlDirection !== null) {
            playerTargetYRef.current = movePongPlayerTarget(
              playerTargetYRef.current,
              mobileControlDirection,
              PONG_MOBILE_CONTROL_SPEED * deltaSeconds,
              dimensionsRef.current.height,
            );
          }

          event = stepPong(state, {
            width: dimensionsRef.current.width,
            height: dimensionsRef.current.height,
            playerTargetY: playerTargetYRef.current,
            difficulty: currentSettings.difficulty,
            deltaSeconds,
            blastEnabled: blastEnabledRef.current,
            blastArmed: blastArmedRef.current,
            blastBars: blastBarsRef.current,
          });

          if (event.playerHit) {
            blastChargeRef.current = event.blastCharge;
            setBlastCharge(event.blastCharge);

            if (event.blastHit) {
              blastArmedRef.current = false;
              setBlastArmed(false);
            }
          }

          if (event.point) {
            setScore((current) => ({
              player: current.player + (event.point === "player" ? 1 : 0),
              cpu: current.cpu + (event.point === "cpu" ? 1 : 0),
            }));
            trailRef.current = [];
          }

          if (currentSettings.trail && !reducedMotionRef.current) {
            trailRef.current = [
              ...trailRef.current,
              { x: state.ball.x, y: state.ball.y },
            ].slice(-14);
          }
          triggerFeedback(event, time);
        }

        const flash = impactFlashRef.current;
        drawPong(
          drawingContext,
          state,
          dimensionsRef.current.width,
          dimensionsRef.current.height,
          trailRef.current,
          currentSettings.trail && !reducedMotionRef.current,
          flash.until > time ? flash.side : null,
          colors,
          blastEnabledRef.current,
          blastArmedRef.current,
          time,
          !reducedMotionRef.current,
        );
      }

      animationFrame = window.requestAnimationFrame(render);
    }

    const resizeObserver = new ResizeObserver(resizeCanvas);
    resizeObserver.observe(stageNode);
    resizeCanvas();
    animationFrame = window.requestAnimationFrame(render);

    return () => {
      window.cancelAnimationFrame(animationFrame);
      resizeObserver.disconnect();
      mobileControlDirectionRef.current = null;
      if (shakeTimerRef.current !== null) {
        window.clearTimeout(shakeTimerRef.current);
      }
    };
  }, []);

  function setEngaged(nextValue: boolean) {
    activeRef.current = nextValue;
    setIsEngaged(nextValue);
  }

  function nudgePlayer(direction: PongControlDirection) {
    playerTargetYRef.current = movePongPlayerTarget(
      playerTargetYRef.current,
      direction,
      PONG_MOBILE_CONTROL_NUDGE,
      dimensionsRef.current.height,
    );
    setEngaged(true);
  }

  function startMobileControl(
    direction: PongControlDirection,
    event: PointerEvent<HTMLButtonElement>,
  ) {
    if (event.pointerType === "mouse" && event.button !== 0) {
      return;
    }

    mobileControlDirectionRef.current = direction;
    nudgePlayer(direction);
    event.currentTarget.setPointerCapture(event.pointerId);
  }

  function stopMobileControl(event: PointerEvent<HTMLButtonElement>) {
    mobileControlDirectionRef.current = null;
    if (event.currentTarget.hasPointerCapture(event.pointerId)) {
      event.currentTarget.releasePointerCapture(event.pointerId);
    }
  }

  function handleMobileControlClick(
    direction: PongControlDirection,
    event: MouseEvent<HTMLButtonElement>,
  ) {
    if (event.detail === 0) {
      nudgePlayer(direction);
    }
  }

  function activateBlast() {
    if (
      !blastEnabledRef.current ||
      blastChargeRef.current < blastBarsRef.current ||
      blastArmedRef.current
    ) {
      return;
    }

    blastArmedRef.current = true;
    setBlastArmed(true);
    setEngaged(true);
  }

  function handleGamePointerDown(event: PointerEvent<HTMLElement>) {
    if (event.pointerType === "mouse" && event.button !== 0) {
      return;
    }

    const target = event.target;
    if (target instanceof HTMLElement && target.closest("button, a, input, select, textarea")) {
      return;
    }

    setEngaged(true);
    activateBlast();
  }

  function handleStagePointerMove(event: PointerEvent<HTMLDivElement>) {
    const rect = event.currentTarget.getBoundingClientRect();
    playerTargetYRef.current = clamp(event.clientY - rect.top, 0, rect.height);
  }

  function handleStageKeyDown(event: KeyboardEvent<HTMLDivElement>) {
    const movement = event.shiftKey ? 36 : 18;
    if (event.key === "ArrowUp" || event.key === "ArrowDown") {
      event.preventDefault();
      playerTargetYRef.current += event.key === "ArrowUp" ? -movement : movement;
      setEngaged(true);
    }
    if (event.key === "Enter" || event.key === " ") {
      event.preventDefault();
      setEngaged(true);
      activateBlast();
    }
    if (event.key === "Escape") {
      setEngaged(false);
    }
  }

  function resetGame() {
    const { width, height } = dimensionsRef.current;
    engineRef.current = createPongState(width, height);
    playerTargetYRef.current = height / 2;
    mobileControlDirectionRef.current = null;
    trailRef.current = [];
    impactFlashRef.current = { side: null, until: 0 };
    blastArmedRef.current = false;
    blastChargeRef.current = 0;
    setBlastArmed(false);
    setBlastCharge(0);
    setScore({ player: 0, cpu: 0 });
  }

  function updateSetting<Key extends keyof PongSettings>(key: Key, value: PongSettings[Key]) {
    setSettings((current) => ({ ...current, [key]: value }));
  }

  function updateBlastBars(value: BlastBars) {
    blastBarsRef.current = value;
    if (!onBlastBarsChange) {
      setInternalBlastBars(value);
    }

    const nextCharge = Math.min(blastChargeRef.current, value);
    blastChargeRef.current = nextCharge;
    if (engineRef.current) {
      engineRef.current.blastCharge = nextCharge;
    }
    setBlastCharge(nextCharge);
    onBlastBarsChange?.(value);
  }

  return (
    <ViComponentSurface
      as="section"
      className={`pong-game ${blastArmed ? "pong-game--blast-armed" : ""} ${className}`.trim()}
      config={visualConfig}
      hoverGlow
      onPointerDown={handleGamePointerDown}
    >
      <div className="pong-game__header">
        <div className="pong-game__copy">
          <span className="section-kicker">A playful systems test</span>
          <h2>Wanna play?</h2>
          <p>
            Move the left paddle with your cursor. Each clean rebound raises the pace and makes the next decision matter.
          </p>
        </div>
        <button
          className="pong-game__settings-button"
          type="button"
          aria-label="Open Pong settings"
          aria-expanded={settingsOpen}
          aria-controls="pong-settings"
          onClick={() => setSettingsOpen((current) => !current)}
        >
          <GearSix size={18} weight="bold" aria-hidden="true" />
        </button>
      </div>

      <div className="pong-game__scoreline" aria-label={`You ${score.player}, CPU ${score.cpu}`}>
        <span>You <strong>{score.player}</strong></span>
        <span className="pong-game__scoreline-rule" aria-hidden="true" />
        <span>CPU <strong>{score.cpu}</strong></span>
      </div>

      {blast ? (
        <div
          className="pong-game__blast-meter"
          aria-label={`Blast Impact charge: ${blastCharge} of ${selectedBlastBars}${blastReady ? ", charged" : ""}`}
        >
          <div className="pong-game__blast-meter-row">
            <span className="pong-game__blast-label">Blast Impact</span>
            <div
              className="pong-game__blast-bars"
              data-ready={blastReady}
              data-bars={selectedBlastBars}
              aria-hidden="true"
            >
              {Array.from({ length: selectedBlastBars }, (_, index) => (
                <span
                  className="pong-game__blast-bar"
                  data-filled={index < blastCharge}
                  key={index}
                />
              ))}
            </div>
            <span className="pong-game__blast-status" data-ready={blastReady}>
              {blastReady ? "Charged" : `${blastCharge}/${selectedBlastBars}`}
            </span>
          </div>
          <span className="pong-game__blast-instruction">
            {blastReady ? "Click anywhere to activate" : `${selectedBlastBars} player impacts to charge`}
          </span>
        </div>
      ) : null}

      <div
        ref={stageRef}
        className="pong-game__stage"
        tabIndex={0}
        data-engaged={isEngaged}
        data-blast-armed={blastArmed}
        aria-label={`Pong game. Move the pointer, use the arrow keys, or use the touch controls below to control your paddle.${blast ? " Click or tap the field when Blast Impact is charged to arm it." : ""}`}
        onPointerEnter={() => setEngaged(true)}
        onPointerLeave={() => setEngaged(false)}
        onPointerMove={handleStagePointerMove}
        onFocus={() => setEngaged(true)}
        onBlur={(event) => {
          if (!event.currentTarget.matches(":hover")) {
            setEngaged(false);
          }
        }}
        onKeyDown={handleStageKeyDown}
      >
        <canvas ref={canvasRef} aria-hidden="true" />
        <span className="pong-game__stage-label">Paddle / rebound</span>

// pong-physics.ts
export type PongDifficulty = "rookie" | "focused" | "sharp";

export type PongPoint = "player" | "cpu";

export type PongState = {
  ball: {
    x: number;
    y: number;
    vx: number;
    vy: number;
    radius: number;
  };
  playerY: number;
  cpuY: number;
  playerHits: number;
  blastCharge: number;
};

export type PongStepConfig = {
  width: number;
  height: number;
  playerTargetY: number;
  difficulty: PongDifficulty;
  deltaSeconds: number;
  blastEnabled?: boolean;
  blastArmed?: boolean;
  blastBars?: BlastBars;
};

export type PongStepEvent = {
  playerHit: boolean;
  cpuHit: boolean;
  point: PongPoint | null;
  highSpeedHit: boolean;
  blastHit: boolean;
  blastCharge: number;
};

export const PONG_PADDLE_WIDTH = 7;
export const PONG_PADDLE_HEIGHT = 44;
export const PONG_MOBILE_CONTROL_SPEED = 260;
export const PONG_MOBILE_CONTROL_NUDGE = 24;
export type PongControlDirection = -1 | 1;
export const BLAST_BAR_OPTIONS = [3, 5, 10] as const;
export type BlastBars = (typeof BLAST_BAR_OPTIONS)[number];
export const DEFAULT_BLAST_BARS: BlastBars = 5;
export const BLAST_MAX_CHARGE = Math.max(...BLAST_BAR_OPTIONS);
export const BLAST_BALL_SPEED = 860;

const PLAYER_X = 18;
const MAX_BALL_SPEED = 430;

const difficultyConfig: Record<
  PongDifficulty,
  { cpuSpeed: number; prediction: number }
> = {
  rookie: { cpuSpeed: 92, prediction: 0.02 },
  focused: { cpuSpeed: 142, prediction: 0.055 },
  sharp: { cpuSpeed: 205, prediction: 0.09 },
};

export function clamp(value: number, min: number, max: number) {
  return Math.min(max, Math.max(min, value));
}

export function movePongPlayerTarget(
  targetY: number,
  direction: PongControlDirection,
  distance: number,
  height: number,
) {
  const minY = PONG_PADDLE_HEIGHT / 2;
  const maxY = Math.max(minY, height - PONG_PADDLE_HEIGHT / 2);

  return clamp(targetY + direction * Math.max(0, distance), minY, maxY);
}

export function createPongState(width: number, height: number): PongState {
  const state: PongState = {
    ball: {
      x: width / 2,
      y: height / 2,
      vx: 132,
      vy: 38,
      radius: 3.5,
    },
    playerY: height / 2,
    cpuY: height / 2,
    playerHits: 0,
    blastCharge: 0,
  };

  return state;
}

export function resetBall(state: PongState, width: number, height: number, direction: -1 | 1) {
  state.ball.x = width / 2;
  state.ball.y = height / 2;
  state.ball.vx = direction * 132;
  state.ball.vy = direction * 38;
  state.playerHits = 0;
}

function moveCpu(state: PongState, config: PongStepConfig, deltaSeconds: number) {
  const { cpuSpeed, prediction } = difficultyConfig[config.difficulty];
  const target = state.ball.y + state.ball.vy * prediction;
  const distance = target - state.cpuY;
  const maxMove = cpuSpeed * deltaSeconds;

  state.cpuY += clamp(distance, -maxMove, maxMove);
  state.cpuY = clamp(
    state.cpuY,
    PONG_PADDLE_HEIGHT / 2,
    config.height - PONG_PADDLE_HEIGHT / 2,
  );
}

function reflectFromPaddle(
  state: PongState,
  paddleY: number,
  direction: -1 | 1,
  blast = false,
) {
  const offset = clamp(
    (state.ball.y - paddleY) / (PONG_PADDLE_HEIGHT / 2),
    -1,
    1,
  );
  const previousSpeed = Math.hypot(state.ball.vx, state.ball.vy);
  const nextSpeed = blast ? BLAST_BALL_SPEED : Math.min(MAX_BALL_SPEED, previousSpeed * 1.045);
  const verticalVelocity = state.ball.vy * 0.88 + offset * 118;
  const angle = clamp(verticalVelocity / nextSpeed, -0.82, 0.82);

  state.ball.vx = direction * Math.sqrt(Math.max(1, nextSpeed ** 2 - (nextSpeed * angle) ** 2));
  state.ball.vy = nextSpeed * angle;
  return nextSpeed;
}

function intersectsPaddle(state: PongState, paddleX: number, paddleY: number) {
  return (
    Math.abs(state.ball.x - paddleX) <= PONG_PADDLE_WIDTH / 2 + state.ball.radius &&
    Math.abs(state.ball.y - paddleY) <= PONG_PADDLE_HEIGHT / 2 + state.ball.radius
  );
}

export function stepPong(state: PongState, config: PongStepConfig): PongStepEvent {
  const deltaSeconds = clamp(config.deltaSeconds, 0, 0.032);
  const blastBars = config.blastBars ?? DEFAULT_BLAST_BARS;
  const event: PongStepEvent = {
    playerHit: false,
    cpuHit: false,
    point: null,
    highSpeedHit: false,
    blastHit: false,
    blastCharge: state.blastCharge,
  };

  state.playerY += (config.playerTargetY - state.playerY) * Math.min(1, deltaSeconds * 18);
  state.playerY = clamp(
    state.playerY,
    PONG_PADDLE_HEIGHT / 2,
    config.height - PONG_PADDLE_HEIGHT / 2,
  );
  moveCpu(state, config, deltaSeconds);

  state.ball.x += state.ball.vx * deltaSeconds;
  state.ball.y += state.ball.vy * deltaSeconds;

  if (state.ball.y - state.ball.radius <= 0) {
    state.ball.y = state.ball.radius;
    state.ball.vy = Math.abs(state.ball.vy);
  } else if (state.ball.y + state.ball.radius >= config.height) {
    state.ball.y = config.height - state.ball.radius;
    state.ball.vy = -Math.abs(state.ball.vy);
  }

  const cpuX = config.width - PLAYER_X;
  if (
    state.ball.vx < 0 &&
    state.ball.x - state.ball.radius <= PLAYER_X + PONG_PADDLE_WIDTH / 2 &&
    intersectsPaddle(state, PLAYER_X, state.playerY)
  ) {
    state.ball.x = PLAYER_X + PONG_PADDLE_WIDTH / 2 + state.ball.radius;
    const blastHit = Boolean(
      config.blastEnabled &&
      config.blastArmed &&
      state.blastCharge >= blastBars,
    );
    const speed = reflectFromPaddle(state, state.playerY, 1, blastHit);
    state.playerHits += 1;
    event.playerHit = true;
    event.blastHit = blastHit;
    if (blastHit) {
      state.blastCharge = 0;
    } else if (config.blastEnabled) {
      state.blastCharge = Math.min(blastBars, state.blastCharge + 1);
    }
    event.blastCharge = state.blastCharge;
    event.highSpeedHit = speed >= 285;
  } else if (
    state.ball.vx > 0 &&
    state.ball.x + state.ball.radius >= cpuX - PONG_PADDLE_WIDTH / 2 &&
    intersectsPaddle(state, cpuX, state.cpuY)
  ) {
    state.ball.x = cpuX - PONG_PADDLE_WIDTH / 2 - state.ball.radius;
    reflectFromPaddle(state, state.cpuY, -1);
    event.cpuHit = true;
  }

  if (state.ball.x + state.ball.radius < 0) {
    event.point = "cpu";
    resetBall(state, config.width, config.height, -1);
  } else if (state.ball.x - state.ball.radius > config.width) {
    event.point = "player";
    resetBall(state, config.width, config.height, 1);
  }

  return event;
}

// shiny-text.tsx
"use client";

import { motion, useAnimationFrame, useMotionValue, useReducedMotion, useTransform } from "motion/react";
import type { ReactNode } from "react";
import { useRef } from "react";

type ShinyTextProps = {
  children: ReactNode;
  className?: string;
  color?: string;
  shineColor?: string;
  speed?: number;
};

export function ShinyText({
  children,
  className = "",
  color = "var(--color-quiet-ink)",
  shineColor = "var(--color-clear-lime)",
  speed = 2.8,
}: ShinyTextProps) {
  const reducedMotion = useReducedMotion();
  const progress = useMotionValue(0);
  const elapsedRef = useRef(0);
  const lastTimeRef = useRef<number | null>(null);
  const duration = speed * 1000;

  useAnimationFrame((time) => {
    if (reducedMotion) {
      lastTimeRef.current = null;
      progress.set(50);
      return;
    }

    if (lastTimeRef.current === null) {
      lastTimeRef.current = time;
      return;
    }

    elapsedRef.current += time - lastTimeRef.current;
    lastTimeRef.current = time;
    progress.set((elapsedRef.current % duration) / duration);
  });

  const backgroundPosition = useTransform(progress, (value) => `${150 - value * 200}% center`);

  return (
    <motion.span
      className={`shiny-text ${className}`.trim()}
      style={{
        backgroundImage: `linear-gradient(110deg, ${color} 0%, ${color} 36%, ${shineColor} 50%, ${color} 64%, ${color} 100%)`,
        backgroundPosition: reducedMotion ? "50% center" : backgroundPosition,
      }}
    >
      {children}
    </motion.span>
  );
}

export function ViComponentSurface({
  as: Component = "div",
  children,
  className = "",
  contentClassName = "",
  config,
  visualProps,
  hoverGlow = false,
  hoverGlowStrength = "medium",
  onPointerDown,

    event.currentTarget.style.setProperty("--vi-glow-y", `${y.toFixed(2)}%`);

    ...createViVisualStyle(resolvedConfig),
