// vi-button.tsx
"use client";

import {
  ArrowUpRight,
  GearSix,
  Plus,
  type Icon,
} from "@phosphor-icons/react";
import {
  forwardRef,
  type ButtonHTMLAttributes,
} from "react";

import {
  createViVisualConfig,
  setViHoverOrigin,
  type ViVisualProps,
} from "./vi-component-config";
import { ViComponentSurface } from "./vi-component-surface";
import "./vi-button.css";

export const VI_BUTTON_ICONS = ["arrow", "plus", "settings"] as const;
export const VI_BUTTON_ICON_POSITIONS = ["none", "left", "right", "only"] as const;
export const VI_BUTTON_SIZES = ["sm", "md", "lg"] as const;
export const VI_BUTTON_VARIANTS = ["active", "standby"] as const;

export type ViButtonIcon = (typeof VI_BUTTON_ICONS)[number];
export type ViButtonIconPosition = (typeof VI_BUTTON_ICON_POSITIONS)[number];
export type ViButtonSize = (typeof VI_BUTTON_SIZES)[number];
export type ViButtonVariant = (typeof VI_BUTTON_VARIANTS)[number];

export type ViButtonProps = Omit<
  ButtonHTMLAttributes<HTMLButtonElement>,
  "children"
> &
  ViVisualProps & {
    bordered?: boolean;
    icon?: ViButtonIcon;
    iconPosition?: ViButtonIconPosition;
    label?: string;
    loading?: boolean;
    size?: ViButtonSize;
    variant?: ViButtonVariant;
  };

const icons: Record<ViButtonIcon, Icon> = {
  arrow: ArrowUpRight,
  plus: Plus,
  settings: GearSix,
};

export const ViButton = forwardRef<HTMLButtonElement, ViButtonProps>(
  function ViButton(
    {
      activeColor,
      activeTextColor,
      "aria-label": ariaLabel,
      bordered = true,
      borderColor,
      className = "",
      disabled = false,
      glass,
      icon = "arrow",
      iconPosition = "right",
      label = "Explore work",
      loading = false,
      onPointerEnter,
      radius = "pill",
      shineColorA,
      shineColorB,
      shinyBorder,
      size = "md",
      standbyColor,
      standbyTextColor,
      type = "button",
      variant = "active",
      ...buttonProps
    },
    ref,
  ) {
    const visual = createViVisualConfig({
      activeColor,
      activeTextColor,
      borderColor,
      glass,
      radius,
      shineColorA,
      shineColorB,
      shinyBorder,
      standbyColor,
      standbyTextColor,
    });
    const IconComponent = icons[icon];
    const iconOnly = iconPosition === "only";
    const showIcon = iconPosition !== "none";
    const buttonLabel = iconOnly ? ariaLabel ?? label : ariaLabel;

    return (
      <ViComponentSurface
        className={`vi-button-shell ${bordered ? "" : "vi-button-shell--borderless"}`.trim()}
        config={visual}
      >
        <button
          {...buttonProps}
          ref={ref}
          className={`vi-button ${className}`.trim()}
          type={type}
          aria-busy={loading || undefined}
          aria-label={buttonLabel}
          data-icon={icon}
          data-icon-only={iconOnly}
          data-loading={loading}
          data-size={size}
          data-variant={variant}
          data-vi-hover
          onPointerEnter={(event) => {
            setViHoverOrigin(event);
            onPointerEnter?.(event);
          }}
          disabled={disabled || loading}
        >
          {loading ? <span className="vi-button__spinner" aria-hidden="true" /> : null}
          {!loading && showIcon && iconPosition === "left" ? (
            <IconComponent className="vi-button__icon" aria-hidden="true" />
          ) : null}
          {!iconOnly ? <span className="vi-button__label">{label}</span> : null}
          {!loading && showIcon && (iconPosition === "right" || iconOnly) ? (
            <IconComponent className="vi-button__icon" aria-hidden="true" />
          ) : null}
        </button>
      </ViComponentSurface>
    );
  },
);

export const VI_RADIUS_OPTIONS = ["none", "sm", "md", "lg", "pill"] as const;

export type ViVisualStyle = CSSProperties & {
  "--vi-active": string;
  "--vi-standby": string;
  "--vi-active-text": string;
  "--vi-standby-text": string;
  "--vi-border": string;
  "--vi-shine-a": string;

  "--color-surface-raised": string;
  "--color-text": string;
  "--color-text-muted": string;
  "--color-quiet-ink": string;
  "--color-line": string;
  "--color-clear-lime": string;
  "--color-accent": string;
  "--color-accent-soft": string;
};

export function createViVisualConfig(
  overrides: ViVisualProps = {},
): ViVisualConfig {
  const definedOverrides = Object.fromEntries(
    Object.entries(overrides).filter(([, value]) => value !== undefined),
  ) as ViVisualProps;

  return { ...DEFAULT_VI_VISUAL_CONFIG, ...definedOverrides };
}

export function resolveViPalette(config: ViVisualConfig) {
  return config.glass ? GLASS_VI_VISUAL_CONFIG : config;
}

export function createViVisualStyle(config: ViVisualConfig): ViVisualStyle {
  const palette = resolveViPalette(config);

  return {
    "--vi-active": palette.activeColor,
    "--vi-standby": palette.standbyColor,
    "--vi-active-text": palette.activeTextColor,
    "--vi-standby-text": palette.standbyTextColor,
    "--vi-border": palette.borderColor,
    "--vi-shine-a": palette.shineColorA,
    "--vi-shine-b": palette.shineColorB,
    "--vi-radius": VI_RADIUS_VALUES[config.radius],
    "--color-bg": palette.standbyColor,
    "--color-surface": palette.standbyColor,
    "--color-surface-raised": `color-mix(in srgb, ${palette.standbyColor} 88%, white)`,
    "--color-text": palette.standbyTextColor,
    "--color-text-muted": `color-mix(in srgb, ${palette.standbyTextColor} 68%, transparent)`,
    "--color-quiet-ink": `color-mix(in srgb, ${palette.standbyTextColor} 46%, transparent)`,
    "--color-line": palette.borderColor,
    "--color-clear-lime": palette.activeColor,
    "--color-accent": palette.activeColor,
    "--color-accent-soft": `color-mix(in srgb, ${palette.activeColor} 11%, transparent)`,
  };
}

export function setViHoverOrigin({ currentTarget, clientX, clientY }: {
  currentTarget: HTMLElement;
  clientX: number;
  clientY: number;
}) {
  const { left, top, width, height } = currentTarget.getBoundingClientRect();
  const x = width ? Math.min(1, Math.max(0, (clientX - left) / width)) : 0.5;
  const y = height ? Math.min(1, Math.max(0, (clientY - top) / height)) : 0.5;

  currentTarget.style.setProperty("--vi-hover-x", (x * 100) + "%");
  currentTarget.style.setProperty("--vi-hover-y", (y * 100) + "%");
}

// vi-component-surface.tsx
"use client";

import type {
  CSSProperties,
  ElementType,
  PointerEvent,
  PointerEventHandler,
  ReactNode,
} from "react";

import {
  createViVisualConfig,
  createViVisualStyle,
  type ViVisualConfig,
  type ViVisualProps,
} from "./vi-component-config";

type ViSurfaceStyle = CSSProperties & {
  "--vi-glow-x": string;
  "--vi-glow-y": string;
};

type ViComponentSurfaceProps = {
  as?: ElementType;
  children: ReactNode;
  className?: string;
  contentClassName?: string;
  config?: ViVisualConfig;
  visualProps?: ViVisualProps;
  hoverGlow?: boolean;
  hoverGlowStrength?: "soft" | "medium" | "strong";
  onPointerDown?: PointerEventHandler<HTMLElement>;
};

export function ViComponentSurface({
  as: Component = "div",
  children,
  className = "",
  contentClassName = "",
  config,
  visualProps,
  hoverGlow = false,
  hoverGlowStrength = "medium",
  onPointerDown,
}: ViComponentSurfaceProps) {
  const resolvedConfig = config ?? createViVisualConfig(visualProps);

  function handlePointerMove(event: PointerEvent<HTMLElement>) {
    const rect = event.currentTarget.getBoundingClientRect();
    const x = Math.min(100, Math.max(0, ((event.clientX - rect.left) / rect.width) * 100));
    const y = Math.min(100, Math.max(0, ((event.clientY - rect.top) / rect.height) * 100));

    event.currentTarget.style.setProperty("--vi-glow-x", `${x.toFixed(2)}%`);
    event.currentTarget.style.setProperty("--vi-glow-y", `${y.toFixed(2)}%`);
  }

  const style: ViSurfaceStyle = {
    ...createViVisualStyle(resolvedConfig),
    "--vi-glow-x": "50%",
    "--vi-glow-y": "50%",
  };

  return (
    <Component
      className={`vi-component-surface ${className}`.trim()}
      data-glass={resolvedConfig.glass}
      data-hover-glow={hoverGlow}
      data-hover-glow-strength={hoverGlowStrength}
      data-shiny={resolvedConfig.shinyBorder}
      onPointerDown={onPointerDown}
      onPointerMove={hoverGlow ? handlePointerMove : undefined}
      style={style}
    >
      <span className="vi-component-surface__shine" aria-hidden="true" />
      <span className="vi-component-surface__hover-glow" aria-hidden="true" />
      <div className={`vi-component-surface__content ${contentClassName}`.trim()}>
        {children}
      </div>
    </Component>
  );
}
