// vi-color-picker.tsx
"use client";

import { CaretDown, Palette } from "@phosphor-icons/react";
import { AnimatePresence, motion, useReducedMotion } from "motion/react";
import {
  useId,
  useRef,
  useState,
  type FocusEvent,
  type KeyboardEvent,
} from "react";

import {
  createViVisualConfig,
  setViHoverOrigin,
  createViVisualStyle,
  type ViVisualProps,
} from "./vi-component-config";
import { ViComponentSurface } from "./vi-component-surface";
import {
  formatHexColor,
  normalizeHexColor,
} from "./vi-color-picker-utils";
import "./vi-color-picker.css";

export const VI_COLOR_PICKER_SIZES = ["sm", "md", "lg"] as const;
export type ViColorPickerSize = (typeof VI_COLOR_PICKER_SIZES)[number];

export type ViColorPickerProps = ViVisualProps & {
  className?: string;
  closeOnSelect?: boolean;
  defaultValue?: string;
  disabled?: boolean;
  id?: string;
  label?: string;
  name?: string;
  onValueChange?: (value: string) => void;
  showInput?: boolean;
  showSwatches?: boolean;
  size?: ViColorPickerSize;
  swatches?: readonly string[];
  value?: string;
};

const DEFAULT_SWATCHES = [
  "#c3f36b",
  "#f3f8f0",
  "#9aaa98",
  "#ff8095",
  "#7dd3fc",
  "#c4b5fd",
] as const;

export function ViColorPicker({
  activeColor,
  activeTextColor,
  borderColor,
  className = "",
  closeOnSelect = false,
  defaultValue = "#c3f36b",
  disabled = false,
  glass,
  id,
  label = "Accent color",
  name,
  onValueChange,
  radius = "md",
  shineColorA,
  shineColorB,
  shinyBorder,
  showInput = true,
  showSwatches = true,
  size = "md",
  standbyColor,
  standbyTextColor,
  swatches = DEFAULT_SWATCHES,
  value,
}: ViColorPickerProps) {
  const generatedId = useId();
  const pickerId = id ?? generatedId;
  const popoverId = `${pickerId}-popover`;
  const triggerRef = useRef<HTMLButtonElement>(null);
  const reducedMotion = useReducedMotion();
  const initialColor = normalizeHexColor(defaultValue) ?? "#c3f36b";
  const [internalValue, setInternalValue] = useState(initialColor);
  const selectedColor = normalizeHexColor(value ?? internalValue) ?? initialColor;
  const [draft, setDraft] = useState({
    source: selectedColor,
    value: selectedColor,
  });
  const [open, setOpen] = useState(false);
  const draftValue = draft.source === selectedColor ? draft.value : selectedColor;
  const draftValid = normalizeHexColor(draftValue) !== null;
  const visual = createViVisualConfig({
    activeColor,
    activeTextColor,
    borderColor,
    glass,
    radius,
    shineColorA,
    shineColorB,
    shinyBorder,
    standbyColor,
    standbyTextColor,
  });

  function chooseColor(nextValue: string, close = closeOnSelect) {
    const normalized = normalizeHexColor(nextValue);
    if (!normalized) return false;

    if (value === undefined) setInternalValue(normalized);
    setDraft({ source: normalized, value: normalized });
    onValueChange?.(normalized);

    if (close) {
      setOpen(false);
      triggerRef.current?.focus();
    }

    return true;
  }

  function commitDraft() {
    if (!chooseColor(draftValue, closeOnSelect)) {
      setDraft({ source: selectedColor, value: selectedColor });
    }
  }

  function handleDraftKeyDown(event: KeyboardEvent<HTMLInputElement>) {
    if (event.key === "Enter") {
      event.preventDefault();
      commitDraft();
    } else if (event.key === "Escape") {
      event.preventDefault();
      setDraft({ source: selectedColor, value: selectedColor });
      setOpen(false);
      triggerRef.current?.focus();
    }
  }

  function handleBlur(event: FocusEvent<HTMLDivElement>) {
    if (!event.currentTarget.contains(event.relatedTarget)) setOpen(false);
  }

  return (
    <div
      className={`vi-color-picker ${className}`.trim()}
      data-disabled={disabled}
      data-size={size}
      onBlur={handleBlur}
      style={createViVisualStyle(visual)}
    >
      <span className="vi-color-picker__label" id={`${pickerId}-label`}>
        {label}
      </span>
      <ViComponentSurface className="vi-color-picker__trigger-surface" config={visual}>
        <button
          ref={triggerRef}
          className="vi-color-picker__trigger"
          data-vi-hover
          onPointerEnter={setViHoverOrigin}
          type="button"
          aria-controls={popoverId}
          aria-expanded={open}
          aria-haspopup="dialog"
          aria-labelledby={`${pickerId}-label ${pickerId}-value`}
          disabled={disabled}
          onClick={() => setOpen((current) => !current)}
        >
          <span className="vi-color-picker__chip" style={{ backgroundColor: selectedColor }} aria-hidden="true" />
          <span id={`${pickerId}-value`}>{formatHexColor(selectedColor)}</span>
          <CaretDown className="vi-color-picker__caret" data-open={open} aria-hidden="true" />
        </button>
      </ViComponentSurface>

      <AnimatePresence>
        {open ? (
          <motion.div
            className="vi-color-picker__popover"
            initial={reducedMotion ? { opacity: 0 } : { opacity: 0, y: -6, scale: 0.985 }}
            animate={{ opacity: 1, y: 0, scale: 1 }}
            exit={reducedMotion ? { opacity: 0 } : { opacity: 0, y: -4, scale: 0.99 }}
            transition={{ duration: reducedMotion ? 0.08 : 0.16, ease: [0.23, 1, 0.32, 1] }}
          >
            <ViComponentSurface className="vi-color-picker__panel-surface" config={visual}>
              <div
                className="vi-color-picker__panel"
                id={popoverId}
                role="dialog"
                aria-label={`${label} controls`}
              >
                <label className="vi-color-picker__native">
                  <span>
                    <Palette aria-hidden="true" />
                    Custom color
                  </span>
                  <input
                    type="color"
                    value={selectedColor}
                    disabled={disabled}
                    onChange={(event) => chooseColor(event.target.value)}
                  />
                </label>

                {showInput ? (
                  <label className="vi-color-picker__hex">
                    <span>Hex value</span>
                    <input
                      type="text"
                      value={draftValue.toUpperCase()}
                      aria-invalid={!draftValid}
                      autoComplete="off"
                      maxLength={7}
                      spellCheck={false}
                      onBlur={commitDraft}
                      onChange={(event) => setDraft({
                        source: selectedColor,
                        value: event.target.value,
                      })}
                      onKeyDown={handleDraftKeyDown}
                    />
                  </label>
                ) : null}

                {showSwatches ? (
                  <div className="vi-color-picker__swatches" role="group" aria-label="Color swatches">
                    {swatches.map((swatch) => {
                      const normalized = normalizeHexColor(swatch);
                      if (!normalized) return null;

                      return (
                        <button
                          key={normalized}
                          className="vi-color-picker__swatch"
                          data-vi-hover
                          onPointerEnter={setViHoverOrigin}
                          type="button"
                          aria-label={`Select ${formatHexColor(normalized)}`}
                          aria-pressed={selectedColor === normalized}
                          style={{ backgroundColor: normalized }}
                          onMouseDown={(event) => event.preventDefault()}
                          onClick={() => chooseColor(normalized)}
                        />
                      );
                    })}
                  </div>
                ) : null}
              </div>
            </ViComponentSurface>
          </motion.div>
        ) : null}
      </AnimatePresence>

      {name ? <input type="hidden" name={name} value={selectedColor} /> : null}
    </div>
  );
}

// vi-color-picker-utils.ts
export function normalizeHexColor(value: string) {
  const compact = value.trim().replace(/^#/, "");

  if (/^[0-9a-f]{3}$/i.test(compact)) {
    return `#${compact.split("").map((character) => character.repeat(2)).join("")}`.toLowerCase();
  }

  if (/^[0-9a-f]{6}$/i.test(compact)) {
    return `#${compact}`.toLowerCase();
  }

  return null;
}

export function formatHexColor(value: string) {
  return (normalizeHexColor(value) ?? "#000000").toUpperCase();
}

export function hexToRgb(value: string) {
  const normalized = normalizeHexColor(value);
  if (!normalized) return null;

  return {
    red: Number.parseInt(normalized.slice(1, 3), 16),
    green: Number.parseInt(normalized.slice(3, 5), 16),
    blue: Number.parseInt(normalized.slice(5, 7), 16),
  };
}

export function createViVisualConfig(
  overrides: ViVisualProps = {},
): ViVisualConfig {
  const definedOverrides = Object.fromEntries(
    Object.entries(overrides).filter(([, value]) => value !== undefined),
  ) as ViVisualProps;

  return { ...DEFAULT_VI_VISUAL_CONFIG, ...definedOverrides };
}

export function resolveViPalette(config: ViVisualConfig) {
  return config.glass ? GLASS_VI_VISUAL_CONFIG : config;
}

export function createViVisualStyle(config: ViVisualConfig): ViVisualStyle {
  const palette = resolveViPalette(config);

  return {
    "--vi-active": palette.activeColor,
    "--vi-standby": palette.standbyColor,
    "--vi-active-text": palette.activeTextColor,
    "--vi-standby-text": palette.standbyTextColor,
    "--vi-border": palette.borderColor,
    "--vi-shine-a": palette.shineColorA,
    "--vi-shine-b": palette.shineColorB,
    "--vi-radius": VI_RADIUS_VALUES[config.radius],
    "--color-bg": palette.standbyColor,
    "--color-surface": palette.standbyColor,
    "--color-surface-raised": `color-mix(in srgb, ${palette.standbyColor} 88%, white)`,
    "--color-text": palette.standbyTextColor,
    "--color-text-muted": `color-mix(in srgb, ${palette.standbyTextColor} 68%, transparent)`,
    "--color-quiet-ink": `color-mix(in srgb, ${palette.standbyTextColor} 46%, transparent)`,
    "--color-line": palette.borderColor,
    "--color-clear-lime": palette.activeColor,
    "--color-accent": palette.activeColor,
    "--color-accent-soft": `color-mix(in srgb, ${palette.activeColor} 11%, transparent)`,
  };
}

export function setViHoverOrigin({ currentTarget, clientX, clientY }: {
  currentTarget: HTMLElement;
  clientX: number;
  clientY: number;
}) {
  const { left, top, width, height } = currentTarget.getBoundingClientRect();
  const x = width ? Math.min(1, Math.max(0, (clientX - left) / width)) : 0.5;
  const y = height ? Math.min(1, Math.max(0, (clientY - top) / height)) : 0.5;

  currentTarget.style.setProperty("--vi-hover-x", (x * 100) + "%");
  currentTarget.style.setProperty("--vi-hover-y", (y * 100) + "%");
}

// vi-component-surface.tsx
"use client";

import type {
  CSSProperties,
  ElementType,
  PointerEvent,
  PointerEventHandler,
  ReactNode,
} from "react";

import {
  createViVisualConfig,
  createViVisualStyle,
  type ViVisualConfig,
  type ViVisualProps,
} from "./vi-component-config";

type ViSurfaceStyle = CSSProperties & {
  "--vi-glow-x": string;
  "--vi-glow-y": string;
};

type ViComponentSurfaceProps = {
  as?: ElementType;
  children: ReactNode;
  className?: string;
  contentClassName?: string;
  config?: ViVisualConfig;
  visualProps?: ViVisualProps;
  hoverGlow?: boolean;
  hoverGlowStrength?: "soft" | "medium" | "strong";
  onPointerDown?: PointerEventHandler<HTMLElement>;
};

export function ViComponentSurface({
  as: Component = "div",
  children,
  className = "",
  contentClassName = "",
  config,
  visualProps,
  hoverGlow = false,
  hoverGlowStrength = "medium",
  onPointerDown,
}: ViComponentSurfaceProps) {
  const resolvedConfig = config ?? createViVisualConfig(visualProps);

  function handlePointerMove(event: PointerEvent<HTMLElement>) {
    const rect = event.currentTarget.getBoundingClientRect();
    const x = Math.min(100, Math.max(0, ((event.clientX - rect.left) / rect.width) * 100));
    const y = Math.min(100, Math.max(0, ((event.clientY - rect.top) / rect.height) * 100));

    event.currentTarget.style.setProperty("--vi-glow-x", `${x.toFixed(2)}%`);
    event.currentTarget.style.setProperty("--vi-glow-y", `${y.toFixed(2)}%`);
  }

  const style: ViSurfaceStyle = {
    ...createViVisualStyle(resolvedConfig),
    "--vi-glow-x": "50%",
    "--vi-glow-y": "50%",
  };

  return (
    <Component
      className={`vi-component-surface ${className}`.trim()}
      data-glass={resolvedConfig.glass}
      data-hover-glow={hoverGlow}
      data-hover-glow-strength={hoverGlowStrength}
      data-shiny={resolvedConfig.shinyBorder}
      onPointerDown={onPointerDown}
      onPointerMove={hoverGlow ? handlePointerMove : undefined}
      style={style}
    >
      <span className="vi-component-surface__shine" aria-hidden="true" />
      <span className="vi-component-surface__hover-glow" aria-hidden="true" />
      <div className={`vi-component-surface__content ${contentClassName}`.trim()}>
        {children}
      </div>
    </Component>
  );
}
