// vi-date-picker.tsx
"use client";

import {
  CalendarBlank,
  CaretDown,
  CaretLeft,
  CaretRight,
  X,
} from "@phosphor-icons/react";
import { AnimatePresence, motion, useReducedMotion } from "motion/react";
import {
  useEffect,
  useId,
  useRef,
  useState,
  type FocusEvent,
  type KeyboardEvent,
} from "react";

import {
  createViVisualConfig,
  createViVisualStyle,
  setViHoverOrigin,
  type ViVisualProps,
} from "./vi-component-config";
import { ViComponentSurface } from "./vi-component-surface";
import {
  addDays,
  addMonths,
  formatLongDate,
  formatMonthLabel,
  formatSelectedDate,
  getCalendarDays,
  getWeekEdge,
  getWeekdayLabels,
  isSameDay,
  isSameMonth,
  normalizeIsoDate,
  normalizeViDateRange,
  type ViDateRange,
  parseIsoDate,
  toIsoDate,
  type ViDatePickerLocale,
  type ViWeekStartsOn,
} from "./vi-date-picker-utils";
import "./vi-date-picker.css";

export const VI_DATE_PICKER_SIZES = ["sm", "md", "lg"] as const;
export const VI_DATE_PICKER_LOCALES = ["en-US", "pt-BR"] as const;
export const VI_DATE_PICKER_WEEK_STARTS = ["sunday", "monday"] as const;

export type ViDatePickerSize = (typeof VI_DATE_PICKER_SIZES)[number];

export type ViDatePickerProps = ViVisualProps & {
  className?: string;
  clearable?: boolean;
  closeOnSelect?: boolean;
  defaultRange?: ViDateRange;
  defaultValue?: string;
  disabled?: boolean;
  id?: string;
  label?: string;
  locale?: ViDatePickerLocale;
  name?: string;
  onRangeChange?: (range: ViDateRange) => void;
  onValueChange?: (value: string) => void;
  placeholder?: string;
  rangeValue?: ViDateRange;
  selectionMode?: "single" | "range";
  showOutsideDays?: boolean;
  size?: ViDatePickerSize;
  value?: string;
  weekStartsOn?: ViWeekStartsOn;
};

export function ViDatePicker({
  activeColor,
  activeTextColor,
  borderColor,
  className = "",
  clearable = true,
  closeOnSelect = true,
  defaultRange,
  defaultValue = "",
  disabled = false,
  glass,
  id,
  label = "Due date",
  locale = "en-US",
  name,
  onRangeChange,
  onValueChange,
  placeholder = "Choose a date",
  radius = "md",
  rangeValue,
  selectionMode = "single",
  shineColorA,
  shineColorB,
  shinyBorder,
  showOutsideDays = true,
  size = "md",
  standbyColor,
  standbyTextColor,
  value,
  weekStartsOn = "monday",
}: ViDatePickerProps) {
  const generatedId = useId();
  const pickerId = id ?? generatedId;
  const popoverId = `${pickerId}-popover`;
  const monthLabelId = `${pickerId}-month`;
  const triggerRef = useRef<HTMLButtonElement>(null);
  const focusFrame = useRef<number | null>(null);
  const dayRefs = useRef(new Map<string, HTMLButtonElement>());
  const reducedMotion = useReducedMotion();
  const controlled = value !== undefined;
  const initialValue = normalizeIsoDate(defaultValue);
  const [internalValue, setInternalValue] = useState(initialValue);
  const rangeMode = selectionMode === "range";
  const [internalRange, setInternalRange] = useState(() => normalizeViDateRange(defaultRange));
  const range = normalizeViDateRange(rangeValue ?? internalRange);
  const selectedValue = rangeMode ? range.start : normalizeIsoDate(controlled ? value : internalValue);
  const endDate = rangeMode ? parseIsoDate(range.end) : null;
  const selectedDate = selectedValue ? parseIsoDate(selectedValue) : null;
  const initialAnchor = selectedDate ?? new Date();
  const [viewDate, setViewDate] = useState(initialAnchor);
  const [activeDate, setActiveDate] = useState(initialAnchor);
  const [open, setOpen] = useState(false);
  const calendarDays = getCalendarDays(viewDate, weekStartsOn);
  const weekdayLabels = getWeekdayLabels(locale, weekStartsOn);
  const today = new Date();
  const localeCopy = locale === "pt-BR"
    ? { clear: "Limpar", next: "Próximo mês", previous: "Mês anterior" }
    : { clear: "Clear", next: "Next month", previous: "Previous month" };
  const visual = createViVisualConfig({
    activeColor,
    activeTextColor,
    borderColor,
    glass,
    radius,
    shineColorA,
    shineColorB,
    shinyBorder,
    standbyColor,
    standbyTextColor,
  });

  useEffect(() => () => {
    if (focusFrame.current !== null) window.cancelAnimationFrame(focusFrame.current);
  }, []);

  function scheduleDayFocus(date: Date) {
    if (focusFrame.current !== null) window.cancelAnimationFrame(focusFrame.current);
    focusFrame.current = window.requestAnimationFrame(() => {
      focusFrame.current = null;
      dayRefs.current.get(toIsoDate(date))?.focus();
    });
  }

  function focusDay(nextDate: Date) {
    setActiveDate(nextDate);
    if (!isSameMonth(nextDate, viewDate)) setViewDate(nextDate);

    scheduleDayFocus(nextDate);
  }

  function openCalendar() {
    const anchor = selectedDate ?? new Date();
    setViewDate(anchor);
    setActiveDate(anchor);
    setOpen(true);

    scheduleDayFocus(anchor);
  }

  function closeCalendar() {
    if (focusFrame.current !== null) {
      window.cancelAnimationFrame(focusFrame.current);
      focusFrame.current = null;
    }
    setOpen(false);
    triggerRef.current?.focus();
  }

  function updateValue(nextValue: string) {
    if (!controlled) setInternalValue(nextValue);
    onValueChange?.(nextValue);
  }

  function updateRange(nextRange: ViDateRange) {
    if (rangeValue === undefined) setInternalRange(nextRange);
    onRangeChange?.(nextRange);
  }

  function selectDate(nextDate: Date) {
    const dateKey = toIsoDate(nextDate);
    const completingRange = Boolean(range.start && !range.end);
    if (rangeMode) {
      updateRange(completingRange
        ? normalizeViDateRange({ start: range.start, end: dateKey })
        : { start: dateKey, end: "" });
    } else {
      updateValue(dateKey);
    }
    setActiveDate(nextDate);
    setViewDate(nextDate);
    if (closeOnSelect && (!rangeMode || completingRange)) closeCalendar();
    else if (rangeMode) focusDay(nextDate);
  }

  function clearDate() {
    if (rangeMode) updateRange({ start: "", end: "" });
    else updateValue("");
    closeCalendar();
  }

  function moveMonth(amount: number) {
    const nextDate = addMonths(activeDate, amount);
    setActiveDate(nextDate);
    setViewDate(nextDate);
  }

  function handleDayKeyDown(
    event: KeyboardEvent<HTMLButtonElement>,
    date: Date,
  ) {
    let nextDate: Date | null = null;

    if (event.key === "ArrowLeft") nextDate = addDays(date, -1);
    else if (event.key === "ArrowRight") nextDate = addDays(date, 1);
    else if (event.key === "ArrowUp") nextDate = addDays(date, -7);
    else if (event.key === "ArrowDown") nextDate = addDays(date, 7);
    else if (event.key === "Home") nextDate = getWeekEdge(date, "start", weekStartsOn);
    else if (event.key === "End") nextDate = getWeekEdge(date, "end", weekStartsOn);
    else if (event.key === "PageUp") nextDate = addMonths(date, event.shiftKey ? -12 : -1);
    else if (event.key === "PageDown") nextDate = addMonths(date, event.shiftKey ? 12 : 1);
    else if (event.key === "Enter" || event.key === " ") {
      event.preventDefault();
      selectDate(date);
      return;
    } else if (event.key === "Escape") {
      event.preventDefault();
      closeCalendar();
      return;
    }

    if (nextDate) {
      event.preventDefault();
      focusDay(nextDate);
    }
  }

  function handleBlur(event: FocusEvent<HTMLDivElement>) {
    if (!event.currentTarget.contains(event.relatedTarget)) setOpen(false);
  }

  return (
    <div
      className={`vi-date-picker ${className}`.trim()}
      data-disabled={disabled}
      data-size={size}
      onBlur={handleBlur}
      style={createViVisualStyle(visual)}
    >
      <span className="vi-date-picker__label" id={`${pickerId}-label`}>
        {label}
      </span>
      <ViComponentSurface className="vi-date-picker__trigger-surface" config={visual}>
        <button
          ref={triggerRef}
          className="vi-date-picker__trigger"
          data-vi-hover
          onPointerEnter={setViHoverOrigin}
          type="button"
          aria-controls={popoverId}
          aria-expanded={open}
          aria-haspopup="dialog"
          aria-labelledby={`${pickerId}-label ${pickerId}-value`}
          disabled={disabled}
          onClick={() => (open ? setOpen(false) : openCalendar())}
        >
          <CalendarBlank className="vi-date-picker__calendar-icon" aria-hidden="true" />
          <span id={`${pickerId}-value`} data-placeholder={!selectedDate}>
            {selectedDate ? formatSelectedDate(selectedDate, locale) : placeholder}
            {endDate && range.end !== range.start ? " – " + formatSelectedDate(endDate, locale) : null}
          </span>
          <CaretDown className="vi-date-picker__caret" data-open={open} aria-hidden="true" />
        </button>
      </ViComponentSurface>

      <AnimatePresence>
        {open ? (
          <motion.div
            className="vi-date-picker__popover"
            initial={reducedMotion ? { opacity: 0 } : { opacity: 0, y: -6, scale: 0.985 }}
            animate={{ opacity: 1, y: 0, scale: 1 }}
            exit={reducedMotion ? { opacity: 0 } : { opacity: 0, y: -4, scale: 0.99 }}
            transition={{ duration: reducedMotion ? 0.08 : 0.16, ease: [0.23, 1, 0.32, 1] }}
          >
            <ViComponentSurface className="vi-date-picker__panel-surface" config={visual}>
              <div
                className="vi-date-picker__panel"
                id={popoverId}
                role="dialog"
                aria-label={label}
              >
                <div className="vi-date-picker__month-nav">
                  <button
                    className="vi-date-picker__nav-button"
                    data-vi-hover
                    onPointerEnter={setViHoverOrigin}
                    type="button"
                    aria-label={localeCopy.previous}
                    onClick={() => moveMonth(-1)}
                  >
                    <CaretLeft aria-hidden="true" />
                  </button>
                  <strong id={monthLabelId} aria-live="polite">
                    {formatMonthLabel(viewDate, locale)}
                  </strong>
                  <button
                    className="vi-date-picker__nav-button"
                    data-vi-hover
                    onPointerEnter={setViHoverOrigin}
                    type="button"
                    aria-label={localeCopy.next}
                    onClick={() => moveMonth(1)}
                  >
                    <CaretRight aria-hidden="true" />
                  </button>
                </div>

                <div className="vi-date-picker__calendar" role="grid" aria-labelledby={monthLabelId} aria-multiselectable={rangeMode || undefined}>
                  <div className="vi-date-picker__weekdays" role="row">
                    {weekdayLabels.map((weekday) => (
                      <span key={weekday.long} role="columnheader" aria-label={weekday.long}>
                        <span aria-hidden="true">{weekday.short}</span>
                      </span>
                    ))}
                  </div>

                  {Array.from({ length: 6 }, (_, weekIndex) => (
                    <div className="vi-date-picker__week" role="row" key={weekIndex}>
                      {calendarDays.slice(weekIndex * 7, weekIndex * 7 + 7).map((date, dayIndex) => {
                        const dateKey = toIsoDate(date);
                        const outside = !isSameMonth(date, viewDate);

                        if (outside && !showOutsideDays) {
                          return <span className="vi-date-picker__empty-day" role="gridcell" key={dateKey} />;
                        }

                        const endpoint = dateKey === selectedValue || (rangeMode && dateKey === range.end);
                        const between = rangeMode && dateKey > range.start && dateKey < range.end;
                        const inRange = rangeMode && Boolean(range.end) && (endpoint || between);
                        const canJoin = (offset: number) => showOutsideDays || isSameMonth(addDays(date, offset), viewDate);

                        return (
                          <button
                            key={dateKey}
                            ref={(node) => {
                              if (node) dayRefs.current.set(dateKey, node);
                              else dayRefs.current.delete(dateKey);
                            }}
                            className="vi-date-picker__day"
                            type="button"
                            role="gridcell"
                            aria-current={isSameDay(date, today) ? "date" : undefined}
                            aria-label={formatLongDate(date, locale)}
                            aria-selected={endpoint || between}
                            data-outside={outside}
                            data-selected={endpoint}
                            data-range-middle={between}
                            data-range-before={inRange && dayIndex > 0 && dateKey > range.start && canJoin(-1)}
                            data-range-after={inRange && dayIndex < 6 && dateKey < range.end && canJoin(1)}
                            data-vi-hover
                            onPointerEnter={setViHoverOrigin}
                            data-today={isSameDay(date, today)}
                            tabIndex={isSameDay(date, activeDate) ? 0 : -1}
                            onClick={() => selectDate(date)}
                            onKeyDown={(event) => handleDayKeyDown(event, date)}
                          >
                            <span>{date.getDate()}</span>
                          </button>
                        );
                      })}
                    </div>
                  ))}
                </div>

                {clearable && selectedDate ? (
                  <button className="vi-date-picker__clear" data-vi-hover onPointerEnter={setViHoverOrigin} type="button" onClick={clearDate}>
                    <X aria-hidden="true" />
                    {localeCopy.clear}
                  </button>
                ) : null}
              </div>
            </ViComponentSurface>
          </motion.div>
        ) : null}
      </AnimatePresence>

      {name ? <input type="hidden" name={name} value={rangeMode && range.end ? range.start + "/" + range.end : selectedValue} disabled={disabled} /> : null}
    </div>
  );
}

// vi-date-picker-utils.ts
export type ViDatePickerLocale = "en-US" | "pt-BR";
export type ViWeekStartsOn = "sunday" | "monday";

function createLocalDate(year: number, month: number, day: number) {
  return new Date(year, month, day, 12, 0, 0, 0);
}

export function parseIsoDate(value: string) {
  const match = /^(\d{4})-(\d{2})-(\d{2})$/.exec(value.trim());
  if (!match) return null;

  const year = Number(match[1]);
  const month = Number(match[2]) - 1;
  const day = Number(match[3]);
  const date = createLocalDate(year, month, day);

  if (
    date.getFullYear() !== year
    || date.getMonth() !== month
    || date.getDate() !== day
  ) {
    return null;
  }

  return date;
}

export function toIsoDate(date: Date) {
  const year = String(date.getFullYear()).padStart(4, "0");
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");
  return `${year}-${month}-${day}`;
}

export function normalizeIsoDate(value: string | undefined) {
  const parsed = value ? parseIsoDate(value) : null;
  return parsed ? toIsoDate(parsed) : "";
}

export function addDays(date: Date, amount: number) {
  return createLocalDate(date.getFullYear(), date.getMonth(), date.getDate() + amount);
}

export function addMonths(date: Date, amount: number) {
  const targetFirst = createLocalDate(date.getFullYear(), date.getMonth() + amount, 1);
  const targetLast = createLocalDate(
    targetFirst.getFullYear(),
    targetFirst.getMonth() + 1,
    0,
  ).getDate();

  return createLocalDate(
    targetFirst.getFullYear(),
    targetFirst.getMonth(),
    Math.min(date.getDate(), targetLast),
  );
}

export function isSameDay(left: Date, right: Date) {
  return toIsoDate(left) === toIsoDate(right);
}

export function isSameMonth(left: Date, right: Date) {
  return left.getFullYear() === right.getFullYear()
    && left.getMonth() === right.getMonth();
}

export function getCalendarDays(
  viewDate: Date,
  weekStartsOn: ViWeekStartsOn,
) {
  const firstOfMonth = createLocalDate(viewDate.getFullYear(), viewDate.getMonth(), 1);
  const firstWeekday = weekStartsOn === "monday" ? 1 : 0;
  const leadingDays = (firstOfMonth.getDay() - firstWeekday + 7) % 7;
  const firstCell = addDays(firstOfMonth, -leadingDays);

  return Array.from({ length: 42 }, (_, index) => addDays(firstCell, index));
}

export function getWeekdayLabels(
  locale: ViDatePickerLocale,
  weekStartsOn: ViWeekStartsOn,
) {
  const sunday = createLocalDate(2021, 0, 3);
  const offset = weekStartsOn === "monday" ? 1 : 0;
  const narrowFormatter = new Intl.DateTimeFormat(locale, { weekday: "narrow" });
  const longFormatter = new Intl.DateTimeFormat(locale, { weekday: "long" });

  return Array.from({ length: 7 }, (_, index) => {
    const date = addDays(sunday, index + offset);
    return {
      long: longFormatter.format(date),
      short: narrowFormatter.format(date),
    };
  });
}

export function formatSelectedDate(date: Date, locale: ViDatePickerLocale) {
  return new Intl.DateTimeFormat(locale, {
    day: "numeric",
    month: "short",
    year: "numeric",
  }).format(date);
}

export function formatMonthLabel(date: Date, locale: ViDatePickerLocale) {
  const label = new Intl.DateTimeFormat(locale, {
    month: "long",
    year: "numeric",
  }).format(date);

  return `${label.charAt(0).toLocaleUpperCase(locale)}${label.slice(1)}`;
}

export function formatLongDate(date: Date, locale: ViDatePickerLocale) {
  return new Intl.DateTimeFormat(locale, {
    day: "numeric",
    month: "long",
    weekday: "long",
    year: "numeric",
  }).format(date);
}

export function getWeekEdge(
  date: Date,
  edge: "end" | "start",
  weekStartsOn: ViWeekStartsOn,
) {
  const firstWeekday = weekStartsOn === "monday" ? 1 : 0;
  const offset = (date.getDay() - firstWeekday + 7) % 7;
  return addDays(date, edge === "start" ? -offset : 6 - offset);
}

export type ViDateRange = { start: string; end: string };

export function normalizeViDateRange(range?: ViDateRange): ViDateRange {
  const start = normalizeIsoDate(range?.start);
  const end = normalizeIsoDate(range?.end);

  if (!start) return { start: end, end: "" };
  return end && end < start ? { start: end, end: start } : { start, end };
}

export function createViVisualConfig(
  overrides: ViVisualProps = {},

  const definedOverrides = Object.fromEntries(
    Object.entries(overrides).filter(([, value]) => value !== undefined),
  ) as ViVisualProps;

  return { ...DEFAULT_VI_VISUAL_CONFIG, ...definedOverrides };
}

export function resolveViPalette(config: ViVisualConfig) {
  return config.glass ? GLASS_VI_VISUAL_CONFIG : config;
}

export function createViVisualStyle(config: ViVisualConfig): ViVisualStyle {
  const palette = resolveViPalette(config);

  return {
    "--vi-active": palette.activeColor,
    "--vi-standby": palette.standbyColor,
    "--vi-active-text": palette.activeTextColor,
    "--vi-standby-text": palette.standbyTextColor,

    "--vi-shine-a": palette.shineColorA,
    "--vi-shine-b": palette.shineColorB,
    "--vi-radius": VI_RADIUS_VALUES[config.radius],
    "--color-bg": palette.standbyColor,
    "--color-surface": palette.standbyColor,
    "--color-surface-raised": `color-mix(in srgb, ${palette.standbyColor} 88%, white)`,
    "--color-text": palette.standbyTextColor,
    "--color-text-muted": `color-mix(in srgb, ${palette.standbyTextColor} 68%, transparent)`,
    "--color-quiet-ink": `color-mix(in srgb, ${palette.standbyTextColor} 46%, transparent)`,
    "--color-line": palette.borderColor,
    "--color-clear-lime": palette.activeColor,
    "--color-accent": palette.activeColor,
    "--color-accent-soft": `color-mix(in srgb, ${palette.activeColor} 11%, transparent)`,
  };
}

export function setViHoverOrigin({ currentTarget, clientX, clientY }: {
  currentTarget: HTMLElement;
  clientX: number;
  clientY: number;
}) {
  const { left, top, width, height } = currentTarget.getBoundingClientRect();
  const x = width ? Math.min(1, Math.max(0, (clientX - left) / width)) : 0.5;
  const y = height ? Math.min(1, Math.max(0, (clientY - top) / height)) : 0.5;

  currentTarget.style.setProperty("--vi-hover-x", (x * 100) + "%");
  currentTarget.style.setProperty("--vi-hover-y", (y * 100) + "%");
}

// vi-component-surface.tsx
"use client";

import type {
  CSSProperties,
  ElementType,
  PointerEvent,
  PointerEventHandler,
  ReactNode,
} from "react";

import {
  createViVisualConfig,
  createViVisualStyle,
  type ViVisualConfig,
  type ViVisualProps,
} from "./vi-component-config";

type ViSurfaceStyle = CSSProperties & {
  "--vi-glow-x": string;
  "--vi-glow-y": string;
};

type ViComponentSurfaceProps = {
  as?: ElementType;
  children: ReactNode;
  className?: string;
  contentClassName?: string;
  config?: ViVisualConfig;
  visualProps?: ViVisualProps;
  hoverGlow?: boolean;
  hoverGlowStrength?: "soft" | "medium" | "strong";
  onPointerDown?: PointerEventHandler<HTMLElement>;
};

export function ViComponentSurface({
  as: Component = "div",
  children,
  className = "",
  contentClassName = "",
  config,
  visualProps,
  hoverGlow = false,
  hoverGlowStrength = "medium",
  onPointerDown,
}: ViComponentSurfaceProps) {
  const resolvedConfig = config ?? createViVisualConfig(visualProps);

  function handlePointerMove(event: PointerEvent<HTMLElement>) {
    const rect = event.currentTarget.getBoundingClientRect();
    const x = Math.min(100, Math.max(0, ((event.clientX - rect.left) / rect.width) * 100));
    const y = Math.min(100, Math.max(0, ((event.clientY - rect.top) / rect.height) * 100));

    event.currentTarget.style.setProperty("--vi-glow-x", `${x.toFixed(2)}%`);
    event.currentTarget.style.setProperty("--vi-glow-y", `${y.toFixed(2)}%`);
  }

  const style: ViSurfaceStyle = {
    ...createViVisualStyle(resolvedConfig),
    "--vi-glow-x": "50%",
    "--vi-glow-y": "50%",
  };

  return (
    <Component
      className={`vi-component-surface ${className}`.trim()}
      data-glass={resolvedConfig.glass}
      data-hover-glow={hoverGlow}
      data-hover-glow-strength={hoverGlowStrength}
      data-shiny={resolvedConfig.shinyBorder}
      onPointerDown={onPointerDown}
      onPointerMove={hoverGlow ? handlePointerMove : undefined}
      style={style}
    >
      <span className="vi-component-surface__shine" aria-hidden="true" />
      <span className="vi-component-surface__hover-glow" aria-hidden="true" />
      <div className={`vi-component-surface__content ${contentClassName}`.trim()}>
        {children}
      </div>
    </Component>
  );
}
