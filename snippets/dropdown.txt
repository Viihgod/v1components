// vi-dropdown.tsx
"use client";

import { CaretDown, Check } from "@phosphor-icons/react";
import { AnimatePresence, motion, useReducedMotion } from "motion/react";
import {
  useId,
  useRef,
  useState,
  type FocusEvent,
  type KeyboardEvent,
} from "react";

import {
  createViVisualConfig,
  setViHoverOrigin,
  createViVisualStyle,
  type ViVisualProps,
} from "./vi-component-config";
import { ViComponentSurface } from "./vi-component-surface";
import "./vi-dropdown.css";

export const VI_DROPDOWN_HOVER_STYLES = ["fill", "line", "glow"] as const;
export const VI_DROPDOWN_SIZES = ["sm", "md", "lg"] as const;

export type ViDropdownHoverStyle = (typeof VI_DROPDOWN_HOVER_STYLES)[number];
export type ViDropdownSize = (typeof VI_DROPDOWN_SIZES)[number];

export type ViDropdownOption = {
  disabled?: boolean;
  label: string;
  value: string;
};

export type ViDropdownProps = ViVisualProps & {
  className?: string;
  defaultValue?: string;
  disabled?: boolean;
  hoverStyle?: ViDropdownHoverStyle;
  id?: string;
  label?: string;
  name?: string;
  onValueChange?: (value: string) => void;
  options?: readonly ViDropdownOption[];
  placeholder?: string;
  size?: ViDropdownSize;
  value?: string;
};

const DEFAULT_OPTIONS: readonly ViDropdownOption[] = [
  { label: "Design engineering", value: "design" },
  { label: "Product systems", value: "systems" },
  { label: "Motion craft", value: "motion" },
];

export function ViDropdown({
  activeColor,
  activeTextColor,
  borderColor,
  className = "",
  defaultValue = "",
  disabled = false,
  glass,
  hoverStyle = "fill",
  id,
  label = "Discipline",
  name,
  onValueChange,
  options = DEFAULT_OPTIONS,
  placeholder = "Select a discipline",
  radius = "md",
  shineColorA,
  shineColorB,
  shinyBorder,
  size = "md",
  standbyColor,
  standbyTextColor,
  value,
}: ViDropdownProps) {
  const generatedId = useId();
  const dropdownId = id ?? generatedId;
  const listboxId = `${dropdownId}-listbox`;
  const triggerRef = useRef<HTMLButtonElement>(null);
  const reducedMotion = useReducedMotion();
  const [internalValue, setInternalValue] = useState(defaultValue);
  const [open, setOpen] = useState(false);
  const selectedValue = value ?? internalValue;
  const selectedIndex = options.findIndex((option) => option.value === selectedValue);
  const [activeIndex, setActiveIndex] = useState(Math.max(0, selectedIndex));
  const selectedOption = selectedIndex >= 0 ? options[selectedIndex] : undefined;
  const activeOption = options[activeIndex];
  const visual = createViVisualConfig({
    activeColor,
    activeTextColor,
    borderColor,
    glass,
    radius,
    shineColorA,
    shineColorB,
    shinyBorder,
    standbyColor,
    standbyTextColor,
  });

  function firstEnabledIndex(direction: 1 | -1, start: number) {
    if (options.length === 0) return -1;

    for (let offset = 0; offset < options.length; offset += 1) {
      const index = (start + offset * direction + options.length) % options.length;
      if (!options[index]?.disabled) return index;
    }

    return -1;
  }

  function openMenu(preferredIndex = selectedIndex) {
    if (disabled || options.length === 0) return;

    const fallback = firstEnabledIndex(1, 0);
    const nextIndex = preferredIndex >= 0 && !options[preferredIndex]?.disabled
      ? preferredIndex
      : fallback;
    setActiveIndex(Math.max(0, nextIndex));
    setOpen(true);
  }

  function moveActive(direction: 1 | -1) {
    const start = activeIndex + direction;
    const nextIndex = firstEnabledIndex(direction, start);
    if (nextIndex >= 0) setActiveIndex(nextIndex);
  }

  function selectOption(option: ViDropdownOption) {
    if (option.disabled) return;

    if (value === undefined) setInternalValue(option.value);
    onValueChange?.(option.value);
    setOpen(false);
    triggerRef.current?.focus();
  }

  function handleKeyDown(event: KeyboardEvent<HTMLButtonElement>) {
    if (disabled) return;

    if (event.key === "ArrowDown" || event.key === "ArrowUp") {
      event.preventDefault();
      if (!open) {
        openMenu(event.key === "ArrowDown" ? selectedIndex : options.length - 1);
      } else {
        moveActive(event.key === "ArrowDown" ? 1 : -1);
      }
      return;
    }

    if (open && (event.key === "Home" || event.key === "End")) {
      event.preventDefault();
      const nextIndex = firstEnabledIndex(
        event.key === "Home" ? 1 : -1,
        event.key === "Home" ? 0 : options.length - 1,
      );
      if (nextIndex >= 0) setActiveIndex(nextIndex);
      return;
    }

    if (event.key === "Enter" || event.key === " ") {
      event.preventDefault();
      if (open && activeOption) selectOption(activeOption);
      else openMenu();
      return;
    }

    if (event.key === "Escape" && open) {
      event.preventDefault();
      setOpen(false);
    }
  }

  function handleBlur(event: FocusEvent<HTMLDivElement>) {
    if (!event.currentTarget.contains(event.relatedTarget)) setOpen(false);
  }

  return (
    <div
      className={`vi-dropdown ${className}`.trim()}
      data-disabled={disabled}
      data-hover-style={hoverStyle}
      data-size={size}
      onBlur={handleBlur}
      style={createViVisualStyle(visual)}
    >
      <span className="vi-dropdown__label" id={`${dropdownId}-label`}>
        {label}
      </span>
      <ViComponentSurface className="vi-dropdown__trigger-surface" config={visual}>
        <button
          ref={triggerRef}
          className="vi-dropdown__trigger"
          data-vi-hover
          onPointerEnter={setViHoverOrigin}
          type="button"
          role="combobox"
          aria-activedescendant={open && activeOption ? `${dropdownId}-option-${activeIndex}` : undefined}
          aria-controls={listboxId}
          aria-expanded={open}
          aria-haspopup="listbox"
          aria-labelledby={`${dropdownId}-label ${dropdownId}-value`}
          disabled={disabled}
          onClick={() => (open ? setOpen(false) : openMenu())}
          onKeyDown={handleKeyDown}
        >
          <span id={`${dropdownId}-value`} data-placeholder={!selectedOption}>
            {selectedOption?.label ?? placeholder}
          </span>
          <CaretDown className="vi-dropdown__caret" aria-hidden="true" data-open={open} />
        </button>
      </ViComponentSurface>

      <AnimatePresence>
        {open ? (
          <motion.div
            className="vi-dropdown__popover"
            initial={reducedMotion ? { opacity: 0 } : { opacity: 0, y: -6, scale: 0.985 }}
            animate={{ opacity: 1, y: 0, scale: 1 }}
            exit={reducedMotion ? { opacity: 0 } : { opacity: 0, y: -4, scale: 0.99 }}
            transition={{ duration: reducedMotion ? 0.08 : 0.16, ease: [0.23, 1, 0.32, 1] }}
          >
            <ViComponentSurface className="vi-dropdown__menu-surface" config={visual}>
              <div className="vi-dropdown__menu" id={listboxId} role="listbox" aria-labelledby={`${dropdownId}-label`}>
                {options.map((option, index) => (
                  <button
                    key={option.value}
                    className="vi-dropdown__option"
                    data-vi-hover={hoverStyle === "fill" && selectedValue !== option.value ? "" : undefined}
                    onPointerEnter={setViHoverOrigin}
                    id={`${dropdownId}-option-${index}`}
                    type="button"
                    role="option"
                    aria-selected={selectedValue === option.value}
                    data-active={activeIndex === index}
                    disabled={option.disabled}
                    tabIndex={-1}
                    onMouseEnter={() => !option.disabled && setActiveIndex(index)}
                    onMouseDown={(event) => event.preventDefault()}
                    onClick={() => selectOption(option)}
                  >
                    <span>{option.label}</span>
                    <Check className="vi-dropdown__check" aria-hidden="true" />
                  </button>
                ))}
              </div>
            </ViComponentSurface>
          </motion.div>
        ) : null}
      </AnimatePresence>

      {name ? <input type="hidden" name={name} value={selectedValue} /> : null}
    </div>
  );
}

export const VI_RADIUS_OPTIONS = ["none", "sm", "md", "lg", "pill"] as const;

export type ViRadius = (typeof VI_RADIUS_OPTIONS)[number];

export type ViVisualConfig = {

  shineColorB: string;

export const DEFAULT_VI_VISUAL_CONFIG: ViVisualConfig = {

  glass: false,

  standbyColor: "#071007",

export const GLASS_VI_VISUAL_CONFIG = {

export type ViVisualStyle = CSSProperties & {
  "--vi-active": string;
  "--vi-standby": string;
  "--vi-active-text": string;
  "--vi-standby-text": string;
  "--vi-border": string;
  "--vi-shine-a": string;
  "--vi-shine-b": string;
  "--vi-radius": string;
  "--color-bg": string;
  "--color-surface": string;
  "--color-surface-raised": string;
  "--color-text": string;
  "--color-text-muted": string;
  "--color-quiet-ink": string;
  "--color-line": string;
  "--color-clear-lime": string;
  "--color-accent": string;
  "--color-accent-soft": string;
};

export function createViVisualConfig(
  overrides: ViVisualProps = {},
): ViVisualConfig {
  const definedOverrides = Object.fromEntries(
    Object.entries(overrides).filter(([, value]) => value !== undefined),
  ) as ViVisualProps;

  return { ...DEFAULT_VI_VISUAL_CONFIG, ...definedOverrides };
}

export function resolveViPalette(config: ViVisualConfig) {
  return config.glass ? GLASS_VI_VISUAL_CONFIG : config;
}

export function createViVisualStyle(config: ViVisualConfig): ViVisualStyle {
  const palette = resolveViPalette(config);

  return {
    "--vi-active": palette.activeColor,
    "--vi-standby": palette.standbyColor,
    "--vi-active-text": palette.activeTextColor,
    "--vi-standby-text": palette.standbyTextColor,
    "--vi-border": palette.borderColor,
    "--vi-shine-a": palette.shineColorA,
    "--vi-shine-b": palette.shineColorB,
    "--vi-radius": VI_RADIUS_VALUES[config.radius],
    "--color-bg": palette.standbyColor,
    "--color-surface": palette.standbyColor,
    "--color-surface-raised": `color-mix(in srgb, ${palette.standbyColor} 88%, white)`,
    "--color-text": palette.standbyTextColor,
    "--color-text-muted": `color-mix(in srgb, ${palette.standbyTextColor} 68%, transparent)`,
    "--color-quiet-ink": `color-mix(in srgb, ${palette.standbyTextColor} 46%, transparent)`,
    "--color-line": palette.borderColor,
    "--color-clear-lime": palette.activeColor,
    "--color-accent": palette.activeColor,
    "--color-accent-soft": `color-mix(in srgb, ${palette.activeColor} 11%, transparent)`,
  };
}

export function setViHoverOrigin({ currentTarget, clientX, clientY }: {
  currentTarget: HTMLElement;
  clientX: number;
  clientY: number;
}) {
  const { left, top, width, height } = currentTarget.getBoundingClientRect();
  const x = width ? Math.min(1, Math.max(0, (clientX - left) / width)) : 0.5;
  const y = height ? Math.min(1, Math.max(0, (clientY - top) / height)) : 0.5;

  currentTarget.style.setProperty("--vi-hover-x", (x * 100) + "%");
  currentTarget.style.setProperty("--vi-hover-y", (y * 100) + "%");
}

// vi-component-surface.tsx
"use client";

import type {
  CSSProperties,
  ElementType,
  PointerEvent,
  PointerEventHandler,
  ReactNode,
} from "react";

import {
  createViVisualConfig,
  createViVisualStyle,
  type ViVisualConfig,
  type ViVisualProps,
} from "./vi-component-config";

type ViSurfaceStyle = CSSProperties & {
  "--vi-glow-x": string;
  "--vi-glow-y": string;
};

type ViComponentSurfaceProps = {
  as?: ElementType;
  children: ReactNode;
  className?: string;
  contentClassName?: string;
  config?: ViVisualConfig;
  visualProps?: ViVisualProps;
  hoverGlow?: boolean;
  hoverGlowStrength?: "soft" | "medium" | "strong";
  onPointerDown?: PointerEventHandler<HTMLElement>;
};

export function ViComponentSurface({
  as: Component = "div",
  children,
  className = "",
  contentClassName = "",
  config,
  visualProps,
  hoverGlow = false,
  hoverGlowStrength = "medium",
  onPointerDown,
}: ViComponentSurfaceProps) {
  const resolvedConfig = config ?? createViVisualConfig(visualProps);

  function handlePointerMove(event: PointerEvent<HTMLElement>) {
    const rect = event.currentTarget.getBoundingClientRect();
    const x = Math.min(100, Math.max(0, ((event.clientX - rect.left) / rect.width) * 100));
    const y = Math.min(100, Math.max(0, ((event.clientY - rect.top) / rect.height) * 100));

    event.currentTarget.style.setProperty("--vi-glow-x", `${x.toFixed(2)}%`);
    event.currentTarget.style.setProperty("--vi-glow-y", `${y.toFixed(2)}%`);
  }

  const style: ViSurfaceStyle = {
    ...createViVisualStyle(resolvedConfig),
    "--vi-glow-x": "50%",
    "--vi-glow-y": "50%",
  };

  return (
    <Component
      className={`vi-component-surface ${className}`.trim()}
      data-glass={resolvedConfig.glass}
      data-hover-glow={hoverGlow}
      data-hover-glow-strength={hoverGlowStrength}
      data-shiny={resolvedConfig.shinyBorder}
      onPointerDown={onPointerDown}
      onPointerMove={hoverGlow ? handlePointerMove : undefined}
      style={style}
    >
      <span className="vi-component-surface__shine" aria-hidden="true" />
      <span className="vi-component-surface__hover-glow" aria-hidden="true" />
      <div className={`vi-component-surface__content ${contentClassName}`.trim()}>
        {children}
      </div>
    </Component>
  );
}
