"use client";

import {
  forwardRef,
  useEffect,
  useRef,
  useState,
  type ButtonHTMLAttributes,
  type CSSProperties,
  type ReactNode,
} from "react";

import type { ViMotionProps } from "../../core/motion-props";
import {
  createViVisualConfig,
  createViVisualStyle,
  type ViVisualProps,
} from "../../core/visual-props";
import { useViReducedMotion } from "../shared/action-feedback-utils";
import "../shared/b04-manipulation.css";

export type ViHoldToTalkState = "idle" | "holding" | "recording" | "cancelled";

export type ViHoldToTalkProps = Omit<
  ButtonHTMLAttributes<HTMLButtonElement>,
  "children" | "disabled"
> &
  ViVisualProps &
  ViMotionProps & {
    children?: ReactNode;
    disabled?: boolean;
    label: string;
    onCancel?: () => void;
    onEnd?: () => void;
    onStart?: () => void;
    state?: ViHoldToTalkState;
    waveform?: ReactNode;
  };

type TalkStyle = CSSProperties;

export const ViHoldToTalk = forwardRef<HTMLButtonElement, ViHoldToTalkProps>(
  function ViHoldToTalk(
    {
      activeColor,
      activeTextColor,
      borderColor,
      children,
      className = "",
      disabled = false,
      glass,
      label,
      motion = "auto",
      motionConfig,
      onBlur,
      onCancel,
      onEnd,
      onKeyDown,
      onKeyUp,
      onPointerCancel,
      onPointerDown,
      onPointerUp,
      onStart,
      paused,
      radius = "pill",
      shineColorA,
      shineColorB,
      shinyBorder,
      standbyColor,
      standbyTextColor,
      state: stateProp,
      style,
      waveform,
      ...rest
    },
    ref,
  ) {
    const [localState, setLocalState] = useState<ViHoldToTalkState>("idle");
    const activeRef = useRef(false);
    const reducedMotion = useViReducedMotion(motion);
    const visual = createViVisualConfig({
      activeColor,
      activeTextColor,
      borderColor,
      glass,
      radius,
      shineColorA,
      shineColorB,
      shinyBorder,
      standbyColor,
      standbyTextColor,
    });
    const talkStyle: TalkStyle = { ...createViVisualStyle(visual), ...style };
    const state = stateProp ?? localState;
    void motionConfig;
    void paused;

    function start() {
      if (disabled || activeRef.current) return;
      activeRef.current = true;
      setLocalState("holding");
      onStart?.();
    }

    function end() {
      if (!activeRef.current) return;
      activeRef.current = false;
      setLocalState("idle");
      onEnd?.();
    }

    function cancel() {
      if (!activeRef.current) return;
      activeRef.current = false;
      setLocalState("cancelled");
      onCancel?.();
    }

    useEffect(() => () => {
      activeRef.current = false;
    }, []);

    return (
      <button
        {...rest}
        ref={ref}
        type={rest.type ?? "button"}
        className={`vi-b04-control vi-b04-talk vi-b04-focus ${className.trim()}`}
        data-motion={reducedMotion ? "reduced" : "auto"}
        data-state={state}
        disabled={disabled}
        aria-label={label}
        aria-pressed={activeRef.current}
        style={talkStyle}
        onBlur={(event) => {
          cancel();
          onBlur?.(event);
        }}
        onKeyDown={(event) => {
          onKeyDown?.(event);
          if (event.defaultPrevented || event.repeat) return;
          if (event.key === "Enter" || event.key === " ") {
            event.preventDefault();
            start();
          }
        }}
        onKeyUp={(event) => {
          onKeyUp?.(event);
          if (event.key === "Enter" || event.key === " ") {
            event.preventDefault();
            end();
          }
        }}
        onPointerCancel={(event) => {
          cancel();
          onPointerCancel?.(event);
        }}
        onPointerDown={(event) => {
          onPointerDown?.(event);
          if (event.defaultPrevented || (event.button !== 0 && event.button !== -1 && event.button !== undefined)) return;
          event.currentTarget.setPointerCapture?.(event.pointerId);
          start();
        }}
        onPointerUp={(event) => {
          onPointerUp?.(event);
          end();
          if (event.currentTarget.hasPointerCapture?.(event.pointerId)) {
            event.currentTarget.releasePointerCapture(event.pointerId);
          }
        }}
      >
        <span className="vi-b04-talk__indicator" aria-hidden="true" />
        <span className="vi-b04-talk__copy"><strong>{children ?? label}</strong><small>{state === "holding" || state === "recording" ? "Release to send" : "Hold to talk"}</small></span>
        {waveform ? <span className="vi-b04-talk__waveform" aria-hidden="true">{waveform}</span> : null}
      </button>
    );
  },
);

"use client";

import { useCallback, useState, type ReactNode } from "react";

export type ViManipulationOrientation = "horizontal" | "vertical";

export function clamp(value: number, min: number, max: number) {
  if (!Number.isFinite(value)) return min;
  return Math.min(max, Math.max(min, value));
}

export function snapToStep(value: number, min: number, max: number, step: number) {
  const safeStep = Number.isFinite(step) && step > 0 ? step : 1;
  const snapped = min + Math.round((value - min) / safeStep) * safeStep;
  const precision = Math.max(
    0,
    (String(safeStep).split(".")[1]?.length ?? 0) +
      (String(min).split(".")[1]?.length ?? 0),
  );
  return clamp(Number(snapped.toFixed(precision)), min, max);
}

export function normalizeRange(
  value: number,
  min: number,
  max: number,
  step = 1,
) {
  const safeMin = Number.isFinite(min) ? min : 0;
  const safeMax = Number.isFinite(max) && max > safeMin ? max : safeMin + 1;
  return snapToStep(value, safeMin, safeMax, step);
}

export function useControllableNumber(
  controlled: number | undefined,
  defaultValue: number,
  min: number,
  max: number,
  step = 1,
) {
  const normalizedDefault = normalizeRange(defaultValue, min, max, step);
  const [uncontrolled, setUncontrolled] = useState(normalizedDefault);
  const isControlled = controlled !== undefined;
  const value = normalizeRange(isControlled ? controlled : uncontrolled, min, max, step);
  const setValue = useCallback(
    (next: number | ((previous: number) => number)) => {
      setUncontrolled((previous) => {
        const resolved = typeof next === "function" ? next(previous) : next;
        return normalizeRange(resolved, min, max, step);
      });
    },
    [max, min, step],
  );

  return [value, setValue, isControlled] as const;
}

export function useControllableNavigationValue<T>(
  controlled: T | undefined,
  defaultValue: T,
) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  const isControlled = controlled !== undefined;
  return [isControlled ? controlled : uncontrolled, setUncontrolled, isControlled] as const;
}

export type ViManipulationItem = string | {
  content?: ReactNode;
  disabled?: boolean;
  id?: string;
  label: string;
  value?: string;
};

export type ResolvedManipulationItem = {
  content?: ReactNode;
  disabled: boolean;
  id: string;
  label: string;
  value: string;
};

export function resolveManipulationItem(
  item: ViManipulationItem,
  index: number,
): ResolvedManipulationItem {
  if (typeof item === "string") {
    return { disabled: false, id: `item-${index}-${item}`, label: item, value: item };
  }

  const value = item.value ?? item.id ?? item.label;
  return {
    content: item.content,
    disabled: Boolean(item.disabled),
    id: item.id ?? `item-${index}-${value}`,
    label: item.label,
    value,
  };
}

export function firstEnabled<T extends { disabled: boolean }>(items: readonly T[]) {
  return items.findIndex((item) => !item.disabled);
}

export function nextEnabled<T extends { disabled: boolean }>(
  items: readonly T[],
  current: number,
  direction: 1 | -1,
  wrap = true,
) {
  if (!items.length) return -1;
  let index = current;
  for (let attempt = 0; attempt < items.length; attempt += 1) {
    const next = index + direction;
    if (!wrap && (next < 0 || next >= items.length)) return current;
    index = (next + items.length) % items.length;
    if (!items[index]?.disabled) return index;
  }
  return current;
}

export function normalizeChordKey(key: string) {
  const normalized = key.trim().toLowerCase();
  if (normalized === " ") return "space";
  if (normalized === "esc") return "escape";
  if (normalized === "arrowleft") return "left";
  if (normalized === "arrowright") return "right";
  if (normalized === "arrowup") return "up";
  if (normalized === "arrowdown") return "down";
  return normalized;
}

export function eventMatchesChordKey(event: { key: string; code: string }, expected: string) {
  const normalizedExpected = normalizeChordKey(expected);
  const candidates = [event.key, event.code.replace(/^Key|^Digit/, "")];
  return candidates.some((candidate) => normalizeChordKey(candidate) === normalizedExpected);
}

export function percentForRange(value: number, min: number, max: number) {
  if (max <= min) return 0;
  return clamp(((value - min) / (max - min)) * 100, 0, 100);
}

export function releasePointerCapture(element: HTMLElement | null, pointerId: number | null) {
  if (element && pointerId !== null && element.hasPointerCapture?.(pointerId)) {
    element.releasePointerCapture(pointerId);
  }
}

"use client";

import { useEffect, useRef, useState, useSyncExternalStore, type CSSProperties } from "react";

import type { ViMotionMode } from "../../core/motion-props";

/** The small style extension shared by action feedback components. */
export type ViActionStyle = CSSProperties & Record<`--vi-${string}`, string | number>;

export function clamp(value: number, min: number, max: number) {
  if (!Number.isFinite(value)) return min;
  return Math.min(max, Math.max(min, value));
}

export function isViReducedMotion(mode: ViMotionMode = "auto") {
  if (mode === "none") return true;
  return typeof window !== "undefined" &&
    typeof window.matchMedia === "function" &&
    Boolean(window.matchMedia("(prefers-reduced-motion: reduce)")?.matches);
}

function subscribeReducedMotion(onStoreChange: () => void) {
  if (typeof window === "undefined" || typeof window.matchMedia !== "function") {
    return () => {};
  }

  const media = window.matchMedia("(prefers-reduced-motion: reduce)");
  const update = () => onStoreChange();
  media.addEventListener?.("change", update);
  media.addListener?.(update);

  return () => {
    media.removeEventListener?.("change", update);
    media.removeListener?.(update);
  };
}

function getReducedMotionSnapshot() {
  return isViReducedMotion("auto");
}

function getReducedMotionServerSnapshot() {
  return false;
}

function noopSubscribe() {
  return () => {};
}

function alwaysReduced() {
  return true;
}

export function useViReducedMotion(mode: ViMotionMode = "auto") {
  const prefersReducedMotion = useSyncExternalStore(
    mode === "none" ? noopSubscribe : subscribeReducedMotion,
    mode === "none" ? alwaysReduced : getReducedMotionSnapshot,
    mode === "none" ? alwaysReduced : getReducedMotionServerSnapshot,
  );

  return mode === "none" || prefersReducedMotion;
}

export function useControllableBoolean(
  controlled: boolean | undefined,
  defaultValue: boolean,
) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  const value = controlled ?? uncontrolled;

  return [value, setUncontrolled] as const;
}

/** Tracks finite timers so transient feedback cannot outlive its component. */
export function useActionTimeout() {
  const timers = useRef(new Set<number>());

  useEffect(() => {
    const activeTimers = timers.current;
    return () => {
      for (const timer of activeTimers) window.clearTimeout(timer);
      activeTimers.clear();
    };
  }, []);

  return (callback: () => void, delayMs: number) => {
    const timer = window.setTimeout(() => {
      timers.current.delete(timer);
      callback();
    }, delayMs);
    timers.current.add(timer);
    return timer;
  };
}

}
