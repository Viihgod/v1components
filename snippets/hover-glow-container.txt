"use client";

import type { CSSProperties, PointerEvent, ReactNode } from "react";
import "./HoverGlowContainer.css";

export type Radius = "none" | "sm" | "md" | "lg" | "pill";

export type VisualProps = {
  shinyBorder?: boolean;
  glass?: boolean;
  radius?: Radius;
  activeColor?: string;
  standbyColor?: string;
  activeTextColor?: string;
  standbyTextColor?: string;
  borderColor?: string;
  shineColorA?: string;
  shineColorB?: string;
};

type ResolvedVisualProps = Required<VisualProps>;

type VisualCSSProperties = CSSProperties & {
  "--vc-active": string;
  "--vc-standby": string;
  "--vc-active-text": string;
  "--vc-standby-text": string;
  "--vc-border": string;
  "--vc-shine-a": string;
  "--vc-shine-b": string;
  "--vc-radius": string;
  "--vc-glow-x": string;
  "--vc-glow-y": string;
};

const RADIUS: Record<Radius, string> = {
  none: "0px",
  sm: "8px",
  md: "16px",
  lg: "28px",
  pill: "999px",
};

const GLASS_PALETTE = {
  activeColor: "rgba(255, 255, 255, 0.18)",
  standbyColor: "rgba(255, 255, 255, 0.055)",
  activeTextColor: "#071007",
  standbyTextColor: "#f7faf7",
  borderColor: "rgba(255, 255, 255, 0.2)",
  shineColorA: "rgba(255, 255, 255, 0.92)",
  shineColorB: "rgba(255, 255, 255, 0.22)",
};

function visualStyle(props: ResolvedVisualProps): VisualCSSProperties {
  const palette = props.glass ? GLASS_PALETTE : props;

  return {
    "--vc-active": palette.activeColor,
    "--vc-standby": palette.standbyColor,
    "--vc-active-text": palette.activeTextColor,
    "--vc-standby-text": palette.standbyTextColor,
    "--vc-border": palette.borderColor,
    "--vc-shine-a": palette.shineColorA,
    "--vc-shine-b": palette.shineColorB,
    "--vc-radius": RADIUS[props.radius],
    "--vc-glow-x": "50%",
    "--vc-glow-y": "50%",
  };
}

function VisualSurface({
  children,
  className = "",
  glowStrength = "medium",
  hoverGlow = false,
  visual,
}: {
  children: ReactNode;
  className?: string;
  glowStrength?: "soft" | "medium" | "strong";
  hoverGlow?: boolean;
  visual: ResolvedVisualProps;
}) {
  function trackPointer(event: PointerEvent<HTMLDivElement>) {
    const rect = event.currentTarget.getBoundingClientRect();
    const x = Math.min(100, Math.max(0, ((event.clientX - rect.left) / rect.width) * 100));
    const y = Math.min(100, Math.max(0, ((event.clientY - rect.top) / rect.height) * 100));
    event.currentTarget.style.setProperty("--vc-glow-x", x.toFixed(2) + "%");
    event.currentTarget.style.setProperty("--vc-glow-y", y.toFixed(2) + "%");
  }

  return (
    <div
      className={("vc-surface " + className).trim()}
      data-glass={visual.glass}
      data-hover-glow={hoverGlow}
      data-hover-glow-strength={glowStrength}
      data-shiny={visual.shinyBorder}
      onPointerMove={hoverGlow ? trackPointer : undefined}
      style={visualStyle(visual)}
    >
      <span className="vc-surface__shine" aria-hidden="true" />
      <span className="vc-surface__glow" aria-hidden="true" />
      <div className="vc-surface__content">{children}</div>
    </div>
  );
}

const DEFAULT_VISUAL = {
  shinyBorder: true,
  glass: false,
  radius: "none",
  activeColor: "#c3f36b",
  standbyColor: "#071007",
  activeTextColor: "#010600",
  standbyTextColor: "#f3f8f0",
  borderColor: "#233a24",
  shineColorA: "#f6ffe2",
  shineColorB: "#c3f36b",
} satisfies Required<VisualProps>;

export type HoverGlowContainerProps = VisualProps & {
  children?: ReactNode;
  eyebrow?: string;
  title?: string;
  description?: string;
  hoverGlow?: boolean;
  glowStrength?: "soft" | "medium" | "strong";
};

export function HoverGlowContainer({
  children,
  eyebrow = "Selected work",
  title = "A focused surface for important content.",
  description = "The border follows the pointer while the inner glow keeps the response spatial.",
  hoverGlow = true,
  glowStrength = "medium",
  ...visualProps
}: HoverGlowContainerProps) {
  const visual = { ...DEFAULT_VISUAL, ...visualProps };

  return (
    <VisualSurface className="hover-glow-container" glowStrength={glowStrength} hoverGlow={hoverGlow} visual={visual}>
      {children ?? (
        <article>
          <span>{eyebrow}</span>
          <h3>{title}</h3>
          <p>{description}</p>
        </article>
      )}
    </VisualSurface>
  );
}
