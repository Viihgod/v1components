export type ViIndeterminateBarProps = Omit<HTMLAttributes<HTMLDivElement>, "children"> &
  ViVisualProps &
  ViMotionProps & {
    active?: boolean;
    label: string;
    size?: ViIndeterminateBarSize;
  };

export const ViIndeterminateBar = forwardRef<HTMLDivElement, ViIndeterminateBarProps>(
  function ViIndeterminateBar(
    {
      active = true,
      activeColor,
      activeTextColor,
      borderColor,
      className = "",
      glass,
      label,
      motion = "auto",
      motionConfig,
      paused,
      radius = "pill",
      shineColorA,
      shineColorB,
      shinyBorder,
      size = "md",
      standbyColor,
      standbyTextColor,
      style,
      ...rest
    },
    ref,
  ) {
    const loop = useViLoopState(motion, paused, active);
    const visual = createViVisualConfig({ activeColor, activeTextColor, borderColor, glass, radius, shineColorA, shineColorB, shinyBorder, standbyColor, standbyTextColor });
    void motionConfig;

    return (
      <div
        {...rest}
        ref={ref}
        className={`vi-b07-control vi-b07-bar ${className.trim()}`.trim()}
        data-active={active}
        data-animate={loop.shouldAnimate}
        data-motion={loop.reducedMotion ? "reduced" : "auto"}
        data-size={size}
        role="progressbar"
        aria-label={label}
        aria-busy={active}
        style={{ ...createViVisualStyle(visual), ...style }}
      >
        <span className="vi-b07-bar__label">{label}</span>
        <span className="vi-b07-bar__track" aria-hidden="true">
          <span className="vi-b07-bar__fill" />
        </span>
      </div>
    );
  },

"use client";

import { useEffect, useRef, useState, useSyncExternalStore } from "react";

import type { ViMotionMode } from "../../core/motion-props";
import { clamp, useViReducedMotion } from "./action-feedback-utils";

export { clamp, useViReducedMotion };

export type ViProgressFormatter = (value: number, max: number) => string;

export function normalizeMax(value: number | undefined, fallback = 100) {
  const next = Number.isFinite(value) ? Number(value) : fallback;
  return Math.max(Number.EPSILON, next > 0 ? next : fallback);
}

export function normalizeProgress(value: number | undefined, max: number, fallback = 0) {
  return clamp(Number.isFinite(value) ? Number(value) : fallback, 0, max);
}

export function progressRatio(value: number | undefined, max: number, fallback = 0) {
  const safeMax = normalizeMax(max);
  return normalizeProgress(value, safeMax, fallback) / safeMax;
}

export function formatProgress(
  value: number,
  max: number,
  formatter?: ViProgressFormatter,
) {
  return formatter?.(value, max) ?? `${Math.round(progressRatio(value, max) * 100)}%`;
}

export function useViDocumentHidden() {
  return useSyncExternalStore(
    (onStoreChange) => {
      if (typeof document === "undefined") return () => {};
      const update = () => onStoreChange();
      document.addEventListener("visibilitychange", update);
      return () => document.removeEventListener("visibilitychange", update);
    },
    () => typeof document !== "undefined" && document.hidden,
    () => false,
  );
}

export function useViLoopState(
  mode: ViMotionMode,
  paused: boolean | undefined,
  active = true,
) {
  const reducedMotion = useViReducedMotion(mode);
  const documentHidden = useViDocumentHidden();
  return {
    documentHidden,
    reducedMotion,
    shouldAnimate: active && !paused && !documentHidden && !reducedMotion,
  };
}

export function useControllableViValue<T>(controlled: T | undefined, defaultValue: T) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  const isControlled = controlled !== undefined;
  return [isControlled ? controlled : uncontrolled, setUncontrolled, isControlled] as const;
}

export function useMinimumLoading(loading: boolean, minDurationMs: number) {
  const duration = Math.max(0, Number.isFinite(minDurationMs) ? minDurationMs : 0);
  const [visible, setVisible] = useState(loading);
  const startedAtRef = useRef<number | null>(null);
  const timerRef = useRef<number | null>(null);

  useEffect(() => {
    if (timerRef.current !== null) {
      window.clearTimeout(timerRef.current);
      timerRef.current = null;
    }

    if (loading) {
      startedAtRef.current = startedAtRef.current ?? Date.now();
      return;
    }

    const startedAt = startedAtRef.current;
    const elapsed = startedAt === null ? duration : Date.now() - startedAt;
    const remaining = Math.max(0, duration - elapsed);
    if (remaining === 0) {
      startedAtRef.current = null;
      setVisible(false);
      return;
    }

    timerRef.current = window.setTimeout(() => {
      timerRef.current = null;
      startedAtRef.current = null;
      setVisible(false);
    }, remaining);

    return () => {
      if (timerRef.current !== null) {
        window.clearTimeout(timerRef.current);
        timerRef.current = null;
      }
    };
  }, [duration, loading]);

  useEffect(() => () => {
    if (timerRef.current !== null) window.clearTimeout(timerRef.current);
  }, []);

  return loading || visible;
}

export function normalizeCount(value: number | undefined, fallback: number, max = 32) {
  return clamp(Math.round(Number.isFinite(value) ? Number(value) : fallback), 1, max);
}

export function normalizeSize(value: number | string | undefined, fallback: number) {
  if (typeof value === "number" && Number.isFinite(value)) return `${clamp(value, 16, 240)}px`;
  if (typeof value === "string" && value.trim()) return value;
  return `${fallback}px`;
}

export function formatCountdown(milliseconds: number) {
  const seconds = Math.max(0, Math.ceil(milliseconds / 1000));
  const minutes = Math.floor(seconds / 60);
  const remainder = seconds % 60;
  return `${minutes}:${String(remainder).padStart(2, "0")}`;
}

export type ViMotionMode = "auto" | "none";

  stiffness: number;
  damping: number;
  mass: number;
};

export type ViMotionConfig = {
  feedbackDurationMs: number;
  enterDurationMs: number;
  exitDurationMs: number;
  progressDurationMs: number;
  ambientDurationMs: number;
  spring: ViSpringConfig;
};

export type ViMotionProps = {
  motion?: ViMotionMode;
  motionConfig?: Partial<ViMotionConfig>;
  paused?: boolean;
};

export const DEFAULT_VI_MOTION_CONFIG: ViMotionConfig = {
  feedbackDurationMs: 120,
  enterDurationMs: 220,
  exitDurationMs: 160,
  progressDurationMs: 420,
  ambientDurationMs: 2800,
  spring: {
    stiffness: 420,
    damping: 32,
    mass: 1,
  },
};

export const VI_SPRING_PRESETS = {
  snappy: { stiffness: 520, damping: 34, mass: 0.8 },
  gentle: { stiffness: 260, damping: 30, mass: 1 },
  bouncy: { stiffness: 420, damping: 20, mass: 0.85 },
  heavy: { stiffness: 220, damping: 38, mass: 1.35 },
} as const satisfies Record<string, ViSpringConfig>;

export function resolveViMotionConfig(
  overrides: Partial<ViMotionConfig> = {},
): ViMotionConfig {
  return {
    ...DEFAULT_VI_MOTION_CONFIG,
    ...overrides,
    spring: {
      ...DEFAULT_VI_MOTION_CONFIG.spring,
      ...overrides.spring,
    },
  };
}

"use client";

import { useEffect, useRef, useState, useSyncExternalStore, type CSSProperties } from "react";

import type { ViMotionMode } from "../../core/motion-props";

/** The small style extension shared by action feedback components. */
export type ViActionStyle = CSSProperties & Record<`--vi-${string}`, string | number>;

export function clamp(value: number, min: number, max: number) {
  if (!Number.isFinite(value)) return min;
  return Math.min(max, Math.max(min, value));
}

export function isViReducedMotion(mode: ViMotionMode = "auto") {
  if (mode === "none") return true;
  return typeof window !== "undefined" &&
    typeof window.matchMedia === "function" &&
    Boolean(window.matchMedia("(prefers-reduced-motion: reduce)")?.matches);
}

function subscribeReducedMotion(onStoreChange: () => void) {
  if (typeof window === "undefined" || typeof window.matchMedia !== "function") {
    return () => {};
  }

  const media = window.matchMedia("(prefers-reduced-motion: reduce)");
  const update = () => onStoreChange();
  media.addEventListener?.("change", update);
  media.addListener?.(update);

  return () => {
    media.removeEventListener?.("change", update);
    media.removeListener?.(update);
  };
}

function getReducedMotionSnapshot() {
  return isViReducedMotion("auto");
}

function getReducedMotionServerSnapshot() {
  return false;
}

function noopSubscribe() {
  return () => {};
}

function alwaysReduced() {
  return true;
}

export function useViReducedMotion(mode: ViMotionMode = "auto") {
  const prefersReducedMotion = useSyncExternalStore(
    mode === "none" ? noopSubscribe : subscribeReducedMotion,
    mode === "none" ? alwaysReduced : getReducedMotionSnapshot,
    mode === "none" ? alwaysReduced : getReducedMotionServerSnapshot,
  );

  return mode === "none" || prefersReducedMotion;
}

export function useControllableBoolean(
  controlled: boolean | undefined,
  defaultValue: boolean,
) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  const value = controlled ?? uncontrolled;

  return [value, setUncontrolled] as const;
}

/** Tracks finite timers so transient feedback cannot outlive its component. */
export function useActionTimeout() {
  const timers = useRef(new Set<number>());

  useEffect(() => {
    const activeTimers = timers.current;
    return () => {
      for (const timer of activeTimers) window.clearTimeout(timer);
      activeTimers.clear();
    };
  }, []);

  return (callback: () => void, delayMs: number) => {
    const timer = window.setTimeout(() => {
      timers.current.delete(timer);
      callback();
    }, delayMs);
    timers.current.add(timer);
    return timer;
  };
}
