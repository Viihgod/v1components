"use client";

import {
  forwardRef,
  useId,
  useRef,
  type CSSProperties,
  type HTMLAttributes,
  type KeyboardEvent,
} from "react";

import type { ViMotionProps } from "../../core/motion-props";
import {
  createViVisualConfig,
  createViVisualStyle,
  type ViVisualProps,
} from "../../core/visual-props";
import { useViReducedMotion } from "../shared/action-feedback-utils";
import {
  firstEnabledIndex,
  nextEnabledIndex,
  resolveNavigationAction,
  toCssLength,
  useControllableNavigationBoolean,
  type ViNavigationAction,
} from "../shared/b03-navigation-utils";
import "../shared/b03-navigation.css";

export type ViOrbitalActionMenuProps = Omit<HTMLAttributes<HTMLDivElement>, "children"> &
  ViVisualProps &
  ViMotionProps & {
    actions?: readonly ViNavigationAction[];
    defaultOpen?: boolean;
    disabled?: boolean;
    label?: string;
    onOpenChange?: (open: boolean) => void;
    open?: boolean;
    radius?: number | string;
    startAngle?: number;
  };

const DEFAULT_ACTIONS: readonly ViNavigationAction[] = ["Add", "Edit", "Share", "Delete"];

type OrbitalStyle = CSSProperties & {
  "--vi-orbital-radius": string;
};

export const ViOrbitalActionMenu = forwardRef<HTMLDivElement, ViOrbitalActionMenuProps>(
  function ViOrbitalActionMenu(
    {
      activeColor,
      activeTextColor,
      actions = DEFAULT_ACTIONS,
      borderColor,
      className = "",
      defaultOpen = false,
      disabled = false,
      glass,
      id,
      label = "Open orbital actions",
      motion = "auto",
      motionConfig,
      onOpenChange,
      open: openProp,
      paused,
      radius = "var(--vi-navigation-orbit-radius)",
      shineColorA,
      shineColorB,
      shinyBorder,
      startAngle = -90,
      standbyColor,
      standbyTextColor,
      style,
      ...rest
    },
    ref,
  ) {
    const resolvedActions = actions.map(resolveNavigationAction);
    const [open, setOpen, controlled] = useControllableNavigationBoolean(openProp, defaultOpen);
    const triggerRef = useRef<HTMLButtonElement>(null);
    const actionRefs = useRef<Array<HTMLButtonElement | null>>([]);
    const generatedId = useId();
    const menuId = "vi-orbital-menu-" + (id ?? generatedId).replaceAll(":", "");
    const reducedMotion = useViReducedMotion(motion);
    const visual = createViVisualConfig({
      activeColor,
      activeTextColor,
      borderColor,
      glass,
      radius: "pill",
      shineColorA,
      shineColorB,
      shinyBorder,
      standbyColor,
      standbyTextColor,
    });
    const orbitalStyle: OrbitalStyle = {
      ...createViVisualStyle(visual),
      "--vi-orbital-radius": toCssLength(radius),
      ...style,
    };
    void motionConfig;
    void paused;

    function setOpenValue(next: boolean, restoreFocus = false) {
      if (disabled) return;
      if (!controlled) setOpen(next);
      onOpenChange?.(next);
      if (restoreFocus) triggerRef.current?.focus();
    }

    function selectAction(index: number) {
      const action = resolvedActions[index];
      if (!action || action.disabled || disabled) return;
      action.onSelect?.();
      setOpenValue(false, true);
    }

    function handleMenuKeyDown(event: KeyboardEvent<HTMLButtonElement>, index: number) {
      if (event.key === "Escape") {
        event.preventDefault();
        setOpenValue(false, true);
        return;
      }
      if (event.key === "ArrowRight" || event.key === "ArrowDown") {
        event.preventDefault();
        actionRefs.current[nextEnabledIndex(resolvedActions, index, 1)]?.focus();
      } else if (event.key === "ArrowLeft" || event.key === "ArrowUp") {
        event.preventDefault();
        actionRefs.current[nextEnabledIndex(resolvedActions, index, -1)]?.focus();
      } else if (event.key === "Home") {
        event.preventDefault();
        actionRefs.current[firstEnabledIndex(resolvedActions)]?.focus();
      } else if (event.key === "End") {
        event.preventDefault();
        const last = [...resolvedActions].map((action, actionIndex) => ({ action, actionIndex })).reverse().find(({ action }) => !action.disabled)?.actionIndex ?? 0;
        actionRefs.current[last]?.focus();
      }
    }

    const angleStep = resolvedActions.length ? 360 / resolvedActions.length : 0;

    return (
      <div
        {...rest}
        ref={ref}
        id={id}
        className={"vi-b03-control vi-b03-orbital " + className.trim()}
        data-motion={reducedMotion ? "reduced" : "auto"}
        data-open={open}
        style={orbitalStyle}
      >
        <button
          ref={triggerRef}
          type="button"
          className="vi-b03-orbital__trigger vi-b03-focus"
          aria-controls={menuId}
          aria-expanded={open}
          aria-haspopup="menu"
          disabled={disabled}
          onClick={() => setOpenValue(!open)}
        >
          <span aria-hidden="true">{open ? "×" : "✦"}</span>
          <span className="vi-b03-sr-only">{label}</span>
        </button>
        <div id={menuId} className="vi-b03-orbital__menu" role="menu" aria-label={label} hidden={!open}>
          {resolvedActions.length ? resolvedActions.map((action, index) => (
            <button
              key={action.id}
              ref={(node) => { actionRefs.current[index] = node; }}
              type="button"
              role="menuitem"
              className="vi-b03-orbital__item vi-b03-focus"
              style={{ "--vi-orbital-angle": String(startAngle + angleStep * index) } as CSSProperties}
              aria-label={action.label}
              disabled={disabled || action.disabled}
              tabIndex={index === firstEnabledIndex(resolvedActions) ? 0 : -1}
              onClick={() => selectAction(index)}
              onKeyDown={(event) => handleMenuKeyDown(event, index)}
            >
              {action.icon ?? <span aria-hidden="true">{action.label.slice(0, 1)}</span>}
            </button>
          )) : <span className="vi-b03-empty">No actions available.</span>}
        </div>
      </div>
    );
  },
);

"use client";

import { useState, type ReactNode } from "react";

export type ViNavigationItem = string | {
  content?: ReactNode;
  description?: ReactNode;
  disabled?: boolean;
  id?: string;
  label: string;
  value?: string;
};

export type ViNavigationAction = string | {
  disabled?: boolean;
  icon?: ReactNode;
  id?: string;
  label: string;
  onSelect?: () => void;
  value?: string;
};

export type ResolvedNavigationItem = {
  content?: ReactNode;
  description?: ReactNode;
  disabled: boolean;
  id: string;
  label: string;
  value: string;
};

export type ResolvedNavigationAction = {
  disabled: boolean;
  icon?: ReactNode;
  id: string;
  label: string;
  onSelect?: () => void;
  value: string;
};

export function resolveNavigationItem(
  item: ViNavigationItem,
  index: number,
): ResolvedNavigationItem {
  if (typeof item === "string") {
    return {
      disabled: false,
      id: "item-" + index + "-" + item,
      label: item,
      value: item,
    };
  }

  const value = item.value ?? item.id ?? item.label;
  return {
    content: item.content,
    description: item.description,
    disabled: Boolean(item.disabled),
    id: item.id ?? "item-" + index + "-" + value,
    label: item.label,
    value,
  };
}

export function resolveNavigationAction(
  action: ViNavigationAction,
  index: number,
): ResolvedNavigationAction {
  if (typeof action === "string") {
    return {
      disabled: false,
      id: "action-" + index + "-" + action,
      label: action,
      value: action,
    };
  }

  const value = action.value ?? action.id ?? action.label;
  return {
    disabled: Boolean(action.disabled),
    icon: action.icon,
    id: action.id ?? "action-" + index + "-" + value,
    label: action.label,
    onSelect: action.onSelect,
    value,
  };
}

export function useControllableNavigationValue<T>(
  controlled: T | undefined,
  defaultValue: T,
) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  const isControlled = controlled !== undefined;
  return [isControlled ? controlled : uncontrolled, setUncontrolled, isControlled] as const;
}

export function useControllableNavigationBoolean(
  controlled: boolean | undefined,
  defaultValue: boolean,
) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  return [controlled ?? uncontrolled, setUncontrolled, controlled !== undefined] as const;
}

export function firstEnabledIndex<T extends { disabled: boolean }>(
  items: readonly T[],
  fallback = 0,
) {
  const index = items.findIndex((item) => !item.disabled);
  return index < 0 ? fallback : index;
}

export function nextEnabledIndex<T extends { disabled: boolean }>(
  items: readonly T[],
  current: number,
  direction: 1 | -1,
) {
  if (!items.length) return -1;
  let index = current;
  for (let attempt = 0; attempt < items.length; attempt += 1) {
    index = (index + direction + items.length) % items.length;
    if (!items[index]?.disabled) return index;
  }
  return current;
}

export function getNavigationKeyDirection(
  key: string,
  orientation: "horizontal" | "vertical" = "horizontal",
): 1 | -1 | 0 {
  if (
    (orientation === "horizontal" && key === "ArrowRight") ||
    (orientation === "vertical" && key === "ArrowDown")
  ) return 1;
  if (
    (orientation === "horizontal" && key === "ArrowLeft") ||
    (orientation === "vertical" && key === "ArrowUp")
  ) return -1;
  return 0;
}

export function toCssLength(value: number | string) {
  return typeof value === "number" ? value + "px" : value;
}

export function matchesShortcut(event: KeyboardEvent, shortcut: string) {
  const parts = shortcut
    .split("+")
    .map((part) => part.trim().toLowerCase())
    .filter(Boolean);
  const key = parts.at(-1);
  if (!key) return false;

  const wantsMeta = parts.includes("cmd") || parts.includes("command") || parts.includes("meta") || parts.includes("⌘");
  const wantsCtrl = parts.includes("ctrl") || parts.includes("control");
  const wantsAlt = parts.includes("alt") || parts.includes("option") || parts.includes("⌥");
  const wantsShift = parts.includes("shift") || parts.includes("⇧");

  return (
    event.key.toLowerCase() === key ||
    (key.length === 1 && event.code.toLowerCase() === "key" + key)
  ) &&
    event.metaKey === wantsMeta &&
    event.ctrlKey === wantsCtrl &&
    event.altKey === wantsAlt &&
    event.shiftKey === wantsShift;
}

"use client";

import { useEffect, useRef, useState, useSyncExternalStore, type CSSProperties } from "react";

import type { ViMotionMode } from "../../core/motion-props";

/** The small style extension shared by action feedback components. */
export type ViActionStyle = CSSProperties & Record<`--vi-${string}`, string | number>;

export function clamp(value: number, min: number, max: number) {
  if (!Number.isFinite(value)) return min;
  return Math.min(max, Math.max(min, value));
}

export function isViReducedMotion(mode: ViMotionMode = "auto") {
  if (mode === "none") return true;
  return typeof window !== "undefined" &&
    typeof window.matchMedia === "function" &&
    Boolean(window.matchMedia("(prefers-reduced-motion: reduce)")?.matches);
}

function subscribeReducedMotion(onStoreChange: () => void) {
  if (typeof window === "undefined" || typeof window.matchMedia !== "function") {
    return () => {};
  }

  const media = window.matchMedia("(prefers-reduced-motion: reduce)");
  const update = () => onStoreChange();
  media.addEventListener?.("change", update);
  media.addListener?.(update);

  return () => {
    media.removeEventListener?.("change", update);
    media.removeListener?.(update);
  };
}

function getReducedMotionSnapshot() {
  return isViReducedMotion("auto");
}

function getReducedMotionServerSnapshot() {
  return false;
}

function noopSubscribe() {
  return () => {};
}

function alwaysReduced() {
  return true;
}

export function useViReducedMotion(mode: ViMotionMode = "auto") {
  const prefersReducedMotion = useSyncExternalStore(
    mode === "none" ? noopSubscribe : subscribeReducedMotion,
    mode === "none" ? alwaysReduced : getReducedMotionSnapshot,
    mode === "none" ? alwaysReduced : getReducedMotionServerSnapshot,
  );

  return mode === "none" || prefersReducedMotion;
}

export function useControllableBoolean(
  controlled: boolean | undefined,
  defaultValue: boolean,
) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  const value = controlled ?? uncontrolled;

  return [value, setUncontrolled] as const;
}

/** Tracks finite timers so transient feedback cannot outlive its component. */
export function useActionTimeout() {
  const timers = useRef(new Set<number>());

  useEffect(() => {
    const activeTimers = timers.current;
    return () => {
      for (const timer of activeTimers) window.clearTimeout(timer);
      activeTimers.clear();
    };
  }, []);

  return (callback: () => void, delayMs: number) => {
    const timer = window.setTimeout(() => {
      timers.current.delete(timer);
      callback();
    }, delayMs);
    timers.current.add(timer);
    return timer;
  };
}

  visual: {
    activeColor?: string;
    activeTextColor?: string;

    glass?: boolean;
    radius?: "none" | "sm" | "md" | "lg" | "pill";
    shineColorA?: string;
    shineColorB?: string;
    shinyBorder?: boolean;
    standbyColor?: string;
    standbyTextColor?: string;
  },
) {
  return visual;
}
