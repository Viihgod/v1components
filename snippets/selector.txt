// vi-selector.tsx
"use client";

import { Check } from "@phosphor-icons/react";
import {
  useId,
  useRef,
  useState,
  type KeyboardEvent,
} from "react";

import {
  createViVisualConfig,
  setViHoverOrigin,
  createViVisualStyle,
  type ViVisualProps,
} from "./vi-component-config";
import { ViComponentSurface } from "./vi-component-surface";
import {
  normalizeViSelection,
  serializeViSelection,
  toggleViSelection,
  type ViSelectionMode,
  type ViSelectionValue,
} from "./vi-selector-state";
import "./vi-selector.css";

export const VI_SELECTOR_HOVER_STYLES = ["fill", "line", "glow"] as const;
export const VI_SELECTOR_SIZES = ["sm", "md", "lg"] as const;

export type ViSelectorHoverStyle = (typeof VI_SELECTOR_HOVER_STYLES)[number];
export type ViSelectorSize = (typeof VI_SELECTOR_SIZES)[number];

export type ViSelectorOption = {
  disabled?: boolean;
  label: string;
  value: string;
};

export type ViSelectorProps = ViVisualProps & {
  className?: string;
  defaultValue?: ViSelectionValue;
  disabled?: boolean;
  hoverStyle?: ViSelectorHoverStyle;
  id?: string;
  label?: string;
  name?: string;
  onValueChange?: (value: string | string[]) => void;
  options?: readonly ViSelectorOption[];
  selectionMode?: ViSelectionMode;
  showCheckbox?: boolean;
  size?: ViSelectorSize;
  value?: ViSelectionValue;
};

const DEFAULT_OPTIONS: readonly ViSelectorOption[] = [
  { label: "Design engineering", value: "design" },
  { label: "Product systems", value: "systems" },
  { label: "Motion craft", value: "motion" },
];

export function ViSelector({
  activeColor,
  activeTextColor,
  borderColor,
  className = "",
  defaultValue = ["design", "motion"],
  disabled = false,
  glass,
  hoverStyle = "fill",
  id,
  label = "Capabilities",
  name,
  onValueChange,
  options = DEFAULT_OPTIONS,
  radius = "md",
  selectionMode = "multiple",
  shineColorA,
  shineColorB,
  shinyBorder,
  showCheckbox = true,
  size = "md",
  standbyColor,
  standbyTextColor,
  value,
}: ViSelectorProps) {
  const generatedId = useId();
  const selectorId = id ?? generatedId;
  const optionRefs = useRef<Array<HTMLButtonElement | null>>([]);
  const [internalSelection, setInternalSelection] = useState(() =>
    normalizeViSelection(defaultValue, selectionMode),
  );
  const controlled = value !== undefined;
  const selection = normalizeViSelection(
    controlled ? value : internalSelection,
    selectionMode,
  );
  const visual = createViVisualConfig({
    activeColor,
    activeTextColor,
    borderColor,
    glass,
    radius,
    shineColorA,
    shineColorB,
    shinyBorder,
    standbyColor,
    standbyTextColor,
  });

  function select(option: ViSelectorOption) {
    if (disabled || option.disabled) return;

    const nextSelection = toggleViSelection(selection, option.value, selectionMode);
    if (!controlled) setInternalSelection(nextSelection);
    onValueChange?.(serializeViSelection(nextSelection, selectionMode));
  }

  function moveFocus(index: number, direction: 1 | -1) {
    for (let offset = 1; offset <= options.length; offset += 1) {
      const nextIndex = (index + offset * direction + options.length) % options.length;
      if (!options[nextIndex]?.disabled) {
        optionRefs.current[nextIndex]?.focus();
        if (selectionMode === "single") select(options[nextIndex]);
        return;
      }
    }
  }

  function handleKeyDown(
    event: KeyboardEvent<HTMLButtonElement>,
    index: number,
  ) {
    if (event.key === "ArrowDown" || event.key === "ArrowRight") {
      event.preventDefault();
      moveFocus(index, 1);
    } else if (event.key === "ArrowUp" || event.key === "ArrowLeft") {
      event.preventDefault();
      moveFocus(index, -1);
    }
  }

  return (
    <div
      className={`vi-selector ${className}`.trim()}
      data-disabled={disabled}
      data-hover-style={hoverStyle}
      data-selection-mode={selectionMode}
      data-show-checkbox={showCheckbox}
      data-size={size}
      style={createViVisualStyle(visual)}
    >
      <span className="vi-selector__label" id={`${selectorId}-label`}>
        {label}
      </span>
      <ViComponentSurface className="vi-selector__surface" config={visual}>
        <div
          className="vi-selector__options"
          role={selectionMode === "single" ? "radiogroup" : "group"}
          aria-labelledby={`${selectorId}-label`}
        >
          {options.map((option, index) => {
            const selected = selection.includes(option.value);

            return (
              <button
                key={option.value}
                ref={(node) => {
                  optionRefs.current[index] = node;
                }}
                className="vi-selector__option"
                data-vi-hover={hoverStyle === "fill" && !selected ? "" : undefined}
                onPointerEnter={setViHoverOrigin}
                type="button"
                role={selectionMode === "single" ? "radio" : "checkbox"}
                aria-checked={selected}
                data-selected={selected}
                disabled={disabled || option.disabled}
                onClick={() => select(option)}
                onKeyDown={(event) => handleKeyDown(event, index)}
              >
                <span className="vi-selector__indicator" aria-hidden="true">
                  <Check />
                </span>
                <span>{option.label}</span>
              </button>
            );
          })}
        </div>
      </ViComponentSurface>

      {name
        ? selection.map((selectedValue) => (
            <input key={selectedValue} type="hidden" name={name} value={selectedValue} />
          ))
        : null}
    </div>
  );
}

// vi-selector-state.ts
export type ViSelectionMode = "multiple" | "single";
export type ViSelectionValue = readonly string[] | string;

export function normalizeViSelection(
  value: ViSelectionValue | undefined,
  mode: ViSelectionMode,
) {
  const values = Array.isArray(value) ? value : value ? [value] : [];
  const uniqueValues = [...new Set(values)];

  return mode === "single" ? uniqueValues.slice(0, 1) : uniqueValues;
}

export function toggleViSelection(
  current: readonly string[],
  option: string,
  mode: ViSelectionMode,
) {
  if (mode === "single") return [option];

  return current.includes(option)
    ? current.filter((value) => value !== option)
    : [...current, option];
}

export function serializeViSelection(
  selection: readonly string[],
  mode: ViSelectionMode,
): string | string[] {
  return mode === "single" ? selection[0] ?? "" : [...selection];
}

  "--color-line": string;
  "--color-clear-lime": string;
  "--color-accent": string;
  "--color-accent-soft": string;

export function createViVisualConfig(
  overrides: ViVisualProps = {},
): ViVisualConfig {
  const definedOverrides = Object.fromEntries(
    Object.entries(overrides).filter(([, value]) => value !== undefined),
  ) as ViVisualProps;

  return { ...DEFAULT_VI_VISUAL_CONFIG, ...definedOverrides };
}

export function resolveViPalette(config: ViVisualConfig) {
  return config.glass ? GLASS_VI_VISUAL_CONFIG : config;
}

export function createViVisualStyle(config: ViVisualConfig): ViVisualStyle {
  const palette = resolveViPalette(config);

  return {
    "--vi-active": palette.activeColor,
    "--vi-standby": palette.standbyColor,
    "--vi-active-text": palette.activeTextColor,
    "--vi-standby-text": palette.standbyTextColor,
    "--vi-border": palette.borderColor,
    "--vi-shine-a": palette.shineColorA,
    "--vi-shine-b": palette.shineColorB,
    "--vi-radius": VI_RADIUS_VALUES[config.radius],
    "--color-bg": palette.standbyColor,
    "--color-surface": palette.standbyColor,
    "--color-surface-raised": `color-mix(in srgb, ${palette.standbyColor} 88%, white)`,
    "--color-text": palette.standbyTextColor,
    "--color-text-muted": `color-mix(in srgb, ${palette.standbyTextColor} 68%, transparent)`,
    "--color-quiet-ink": `color-mix(in srgb, ${palette.standbyTextColor} 46%, transparent)`,
    "--color-line": palette.borderColor,
    "--color-clear-lime": palette.activeColor,
    "--color-accent": palette.activeColor,
    "--color-accent-soft": `color-mix(in srgb, ${palette.activeColor} 11%, transparent)`,
  };
}

export function setViHoverOrigin({ currentTarget, clientX, clientY }: {
  currentTarget: HTMLElement;
  clientX: number;
  clientY: number;
}) {
  const { left, top, width, height } = currentTarget.getBoundingClientRect();
  const x = width ? Math.min(1, Math.max(0, (clientX - left) / width)) : 0.5;
  const y = height ? Math.min(1, Math.max(0, (clientY - top) / height)) : 0.5;

  currentTarget.style.setProperty("--vi-hover-x", (x * 100) + "%");
  currentTarget.style.setProperty("--vi-hover-y", (y * 100) + "%");
}

// vi-component-surface.tsx
"use client";

import type {
  CSSProperties,
  ElementType,
  PointerEvent,
  PointerEventHandler,
  ReactNode,
} from "react";

import {
  createViVisualConfig,
  createViVisualStyle,
  type ViVisualConfig,
  type ViVisualProps,
} from "./vi-component-config";

type ViSurfaceStyle = CSSProperties & {
  "--vi-glow-x": string;
  "--vi-glow-y": string;
};

type ViComponentSurfaceProps = {
  as?: ElementType;
  children: ReactNode;
  className?: string;
  contentClassName?: string;
  config?: ViVisualConfig;
  visualProps?: ViVisualProps;
  hoverGlow?: boolean;
  hoverGlowStrength?: "soft" | "medium" | "strong";
  onPointerDown?: PointerEventHandler<HTMLElement>;
};

export function ViComponentSurface({
  as: Component = "div",
  children,
  className = "",
  contentClassName = "",
  config,
  visualProps,
  hoverGlow = false,
  hoverGlowStrength = "medium",
  onPointerDown,
}: ViComponentSurfaceProps) {
  const resolvedConfig = config ?? createViVisualConfig(visualProps);

  function handlePointerMove(event: PointerEvent<HTMLElement>) {
    const rect = event.currentTarget.getBoundingClientRect();
    const x = Math.min(100, Math.max(0, ((event.clientX - rect.left) / rect.width) * 100));
    const y = Math.min(100, Math.max(0, ((event.clientY - rect.top) / rect.height) * 100));

    event.currentTarget.style.setProperty("--vi-glow-x", `${x.toFixed(2)}%`);
    event.currentTarget.style.setProperty("--vi-glow-y", `${y.toFixed(2)}%`);
  }

  const style: ViSurfaceStyle = {
    ...createViVisualStyle(resolvedConfig),
    "--vi-glow-x": "50%",
    "--vi-glow-y": "50%",
  };

  return (
    <Component
      className={`vi-component-surface ${className}`.trim()}
      data-glass={resolvedConfig.glass}
      data-hover-glow={hoverGlow}
      data-hover-glow-strength={hoverGlowStrength}
      data-shiny={resolvedConfig.shinyBorder}
      onPointerDown={onPointerDown}
      onPointerMove={hoverGlow ? handlePointerMove : undefined}
      style={style}
    >
      <span className="vi-component-surface__shine" aria-hidden="true" />
      <span className="vi-component-surface__hover-glow" aria-hidden="true" />
      <div className={`vi-component-surface__content ${contentClassName}`.trim()}>
        {children}
      </div>
    </Component>
  );
}
