"use client";

import { AnimatePresence, motion, useReducedMotion } from "motion/react";
import { useId, useState, type CSSProperties, type PointerEvent, type ReactNode } from "react";
import "./SpringModal.css";

export type Radius = "none" | "sm" | "md" | "lg" | "pill";

export type VisualProps = {
  shinyBorder?: boolean;
  glass?: boolean;
  radius?: Radius;
  activeColor?: string;
  standbyColor?: string;
  activeTextColor?: string;
  standbyTextColor?: string;
  borderColor?: string;
  shineColorA?: string;
  shineColorB?: string;
};

type ResolvedVisualProps = Required<VisualProps>;

type VisualCSSProperties = CSSProperties & {
  "--vc-active": string;
  "--vc-standby": string;
  "--vc-active-text": string;
  "--vc-standby-text": string;
  "--vc-border": string;
  "--vc-shine-a": string;
  "--vc-shine-b": string;
  "--vc-radius": string;
  "--vc-glow-x": string;
  "--vc-glow-y": string;
};

const RADIUS: Record<Radius, string> = {
  none: "0px",
  sm: "8px",
  md: "16px",
  lg: "28px",
  pill: "999px",
};

const GLASS_PALETTE = {
  activeColor: "rgba(255, 255, 255, 0.18)",
  standbyColor: "rgba(255, 255, 255, 0.055)",
  activeTextColor: "#071007",
  standbyTextColor: "#f7faf7",
  borderColor: "rgba(255, 255, 255, 0.2)",
  shineColorA: "rgba(255, 255, 255, 0.92)",
  shineColorB: "rgba(255, 255, 255, 0.22)",
};

function visualStyle(props: ResolvedVisualProps): VisualCSSProperties {
  const palette = props.glass ? GLASS_PALETTE : props;

  return {
    "--vc-active": palette.activeColor,
    "--vc-standby": palette.standbyColor,
    "--vc-active-text": palette.activeTextColor,
    "--vc-standby-text": palette.standbyTextColor,
    "--vc-border": palette.borderColor,
    "--vc-shine-a": palette.shineColorA,
    "--vc-shine-b": palette.shineColorB,
    "--vc-radius": RADIUS[props.radius],
    "--vc-glow-x": "50%",
    "--vc-glow-y": "50%",
  };
}

function VisualSurface({
  children,
  className = "",
  glowStrength = "medium",
  hoverGlow = false,
  visual,
}: {
  children: ReactNode;
  className?: string;
  glowStrength?: "soft" | "medium" | "strong";
  hoverGlow?: boolean;
  visual: ResolvedVisualProps;
}) {
  function trackPointer(event: PointerEvent<HTMLDivElement>) {
    const rect = event.currentTarget.getBoundingClientRect();
    const x = Math.min(100, Math.max(0, ((event.clientX - rect.left) / rect.width) * 100));
    const y = Math.min(100, Math.max(0, ((event.clientY - rect.top) / rect.height) * 100));
    event.currentTarget.style.setProperty("--vc-glow-x", x.toFixed(2) + "%");
    event.currentTarget.style.setProperty("--vc-glow-y", y.toFixed(2) + "%");
  }

  return (
    <div
      className={("vc-surface " + className).trim()}
      data-glass={visual.glass}
      data-hover-glow={hoverGlow}
      data-hover-glow-strength={glowStrength}
      data-shiny={visual.shinyBorder}
      onPointerMove={hoverGlow ? trackPointer : undefined}
      style={visualStyle(visual)}
    >
      <span className="vc-surface__shine" aria-hidden="true" />
      <span className="vc-surface__glow" aria-hidden="true" />
      <div className="vc-surface__content">{children}</div>
    </div>
  );
}

function setViHoverOrigin({ currentTarget, clientX, clientY }: {
  currentTarget: HTMLElement;
  clientX: number;
  clientY: number;
}) {
  const { left, top, width, height } = currentTarget.getBoundingClientRect();
  const x = width ? Math.min(1, Math.max(0, (clientX - left) / width)) : 0.5;
  const y = height ? Math.min(1, Math.max(0, (clientY - top) / height)) : 0.5;

  currentTarget.style.setProperty("--vi-hover-x", (x * 100) + "%");
  currentTarget.style.setProperty("--vi-hover-y", (y * 100) + "%");
}

const DEFAULT_VISUAL = {
  shinyBorder: false,
  glass: false,
  radius: "md",
  activeColor: "#c3f36b",
  standbyColor: "#071007",
  activeTextColor: "#010600",
  standbyTextColor: "#f3f8f0",
  borderColor: "#233a24",
  shineColorA: "#f6ffe2",
  shineColorB: "#c3f36b",
} satisfies Required<VisualProps>;

export type SpringModalProps = VisualProps & {
  triggerLabel?: string;
  eyebrow?: string;
  title?: string;
  closeOnBackdrop?: boolean;
};

export function SpringModal({
  triggerLabel = "Open preview",
  eyebrow = "Preview state",
  title = "Small motion, clear feedback.",
  closeOnBackdrop = true,
  ...visualProps
}: SpringModalProps) {
  const [open, setOpen] = useState(false);
  const titleId = useId();
  const reducedMotion = useReducedMotion();
  const visual = { ...DEFAULT_VISUAL, ...visualProps };

  return (
    <>
      <VisualSurface className="spring-modal__trigger-surface" visual={visual}>
        <button className="spring-modal__trigger" data-vi-hover onPointerEnter={setViHoverOrigin} type="button" onClick={() => setOpen(true)}>
          {triggerLabel}
        </button>
      </VisualSurface>

      <AnimatePresence>
        {open ? (
          <motion.div
            className="spring-modal__backdrop"
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            transition={{ duration: reducedMotion ? 0.1 : 0.2 }}
            onClick={() => closeOnBackdrop && setOpen(false)}
          >
            <motion.div
              initial={reducedMotion ? { opacity: 0 } : { opacity: 0, y: 14, scale: 0.97 }}
              animate={reducedMotion ? { opacity: 1 } : { opacity: 1, y: 0, scale: 1 }}
              exit={reducedMotion ? { opacity: 0 } : { opacity: 0, y: 8, scale: 0.98 }}
              transition={reducedMotion ? { duration: 0.1 } : { type: "spring", stiffness: 360, damping: 28 }}
              onClick={(event) => event.stopPropagation()}
            >
              <VisualSurface className="spring-modal__panel" visual={visual}>
                <div role="dialog" aria-modal="true" aria-labelledby={titleId}>
                  <span className="spring-modal__eyebrow">{eyebrow}</span>
                  <strong id={titleId}>{title}</strong>
                  <button data-vi-hover onPointerEnter={setViHoverOrigin} type="button" onClick={() => setOpen(false)}>Close</button>
                </div>
              </VisualSurface>
            </motion.div>
          </motion.div>
        ) : null}
      </AnimatePresence>
    </>
  );
}
