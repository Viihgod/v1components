import { forwardRef, type ButtonHTMLAttributes, type ReactNode } from "react";

import type { ViMotionProps } from "../../core/motion-props";

import type { ViVisualProps } from "../../core/visual-props";

  clamp,
  createActionVisualProps,
  type ViActionStyle,
  useViReducedMotion,
} from "../shared/action-feedback-utils";

export type ViSquishButtonProps = Omit<
  ButtonHTMLAttributes<HTMLButtonElement>,
  "children"
> &
  ViVisualProps &
  ViMotionProps & {
    children: ReactNode;
    disabled?: boolean;
    pressScaleX?: number;
    pressScaleY?: number;
  };

export const ViSquishButton = forwardRef<HTMLButtonElement, ViSquishButtonProps>(
  function ViSquishButton(
    {
      activeColor,
      activeTextColor,
      borderColor,
      children,
      className = "",
      disabled = false,
      glass,
      motion = "auto",
      motionConfig,
      paused,
      pressScaleX = 0.96,
      pressScaleY = 0.92,
      radius = "md",
      shineColorA,
      shineColorB,
      shinyBorder,
      standbyColor,
      standbyTextColor,
      style,
      ...buttonProps
    },
    ref,
  ) {
    const reducedMotion = useViReducedMotion(motion);
    void motionConfig;
    void paused;

    const visualProps = createActionVisualProps({
      activeColor,
      activeTextColor,
      borderColor,
      glass,
      radius,
      shineColorA,
      shineColorB,
      shinyBorder,
      standbyColor,
      standbyTextColor,
    });
    const surfaceStyle: ViActionStyle = {
      ...style,
      "--vi-squish-x": clamp(pressScaleX, 0.8, 1),
      "--vi-squish-y": clamp(pressScaleY, 0.8, 1),
    };

    return (
      <ViComponentSurface
        {...buttonProps}
        ref={ref}
        as="button"
        className={`vi-squish-button ${className}`.trim()}
        contentAs="span"
        data-motion={reducedMotion ? "reduced" : "auto"}
        disabled={disabled}
        style={surfaceStyle}
        type="button"
        visualProps={visualProps}
      >
        {children}
      </ViComponentSurface>
    );
  },
);

// --- src/components/vicomponents/components/shared/action-feedback-utils.ts ---
"use client";

import { useEffect, useRef, useState, useSyncExternalStore, type CSSProperties } from "react";

import type { ViMotionMode } from "../../core/motion-props";

/** The small style extension shared by action feedback components. */
export type ViActionStyle = CSSProperties & Record<`--vi-${string}`, string | number>;

export function clamp(value: number, min: number, max: number) {
  if (!Number.isFinite(value)) return min;
  return Math.min(max, Math.max(min, value));
}

export function isViReducedMotion(mode: ViMotionMode = "auto") {
  if (mode === "none") return true;
  return typeof window !== "undefined" &&
    typeof window.matchMedia === "function" &&
    Boolean(window.matchMedia("(prefers-reduced-motion: reduce)")?.matches);
}

function subscribeReducedMotion(onStoreChange: () => void) {
  if (typeof window === "undefined" || typeof window.matchMedia !== "function") {
    return () => {};
  }

  const media = window.matchMedia("(prefers-reduced-motion: reduce)");
  const update = () => onStoreChange();
  media.addEventListener?.("change", update);
  media.addListener?.(update);

  return () => {
    media.removeEventListener?.("change", update);
    media.removeListener?.(update);
  };
}

function getReducedMotionSnapshot() {
  return isViReducedMotion("auto");
}

function getReducedMotionServerSnapshot() {
  return false;
}

function noopSubscribe() {
  return () => {};
}

function alwaysReduced() {
  return true;
}

export function useViReducedMotion(mode: ViMotionMode = "auto") {
  const prefersReducedMotion = useSyncExternalStore(
    mode === "none" ? noopSubscribe : subscribeReducedMotion,
    mode === "none" ? alwaysReduced : getReducedMotionSnapshot,
    mode === "none" ? alwaysReduced : getReducedMotionServerSnapshot,
  );

  return mode === "none" || prefersReducedMotion;
}

export function useControllableBoolean(
  controlled: boolean | undefined,
  defaultValue: boolean,
) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  const value = controlled ?? uncontrolled;

  return [value, setUncontrolled] as const;
}

/** Tracks finite timers so transient feedback cannot outlive its component. */
export function useActionTimeout() {
  const timers = useRef(new Set<number>());

  useEffect(() => {
    const activeTimers = timers.current;
    return () => {
      for (const timer of activeTimers) window.clearTimeout(timer);
      activeTimers.clear();
    };
  }, []);

  return (callback: () => void, delayMs: number) => {
    const timer = window.setTimeout(() => {
      timers.current.delete(timer);
      callback();
    }, delayMs);
    timers.current.add(timer);
    return timer;
  };
}

export function createActionVisualProps(
  visual: {
    activeColor?: string;
    activeTextColor?: string;
    borderColor?: string;
    glass?: boolean;
    radius?: "none" | "sm" | "md" | "lg" | "pill";
    shineColorA?: string;
    shineColorB?: string;
    shinyBorder?: boolean;
    standbyColor?: string;
    standbyTextColor?: string;
  },
) {
  return visual;
}

"use client";

import {
  forwardRef,
  type ComponentPropsWithoutRef,
  type ComponentPropsWithRef,
  type CSSProperties,
  type ElementType,
  type ForwardRefRenderFunction,
  type PointerEvent,
  type PointerEventHandler,
  type ReactElement,
  type ReactNode,
} from "react";

import {
  createViVisualConfig,
  createViVisualStyle,
  type ViVisualConfig,
  type ViVisualProps,
} from "./visual-props";
import "./vi-component-surface.css";

type ViSurfaceStyle = CSSProperties & {
  "--vi-glow-x": string;
  "--vi-glow-y": string;
};

type ViComponentSurfaceOwnProps = {
  children: ReactNode;
  className?: string;
  config?: ViVisualConfig;
  contentAs?: "div" | "span";
  contentClassName?: string;
  hoverGlow?: boolean;
  hoverGlowStrength?: "soft" | "medium" | "strong";
  onPointerDown?: PointerEventHandler<HTMLElement>;
  onPointerMove?: PointerEventHandler<HTMLElement>;
  visualProps?: ViVisualProps;
};

export type ViComponentSurfaceProps<T extends ElementType = "div"> =
  ViComponentSurfaceOwnProps &
    Omit<ComponentPropsWithoutRef<T>, keyof ViComponentSurfaceOwnProps | "as" | "children" | "style"> & {
      as?: T;
      style?: CSSProperties;
    };

function ViComponentSurfaceImpl<T extends ElementType = "div">(
  {
    as,
    children,
    className = "",
    config,
    contentAs = "div",
    contentClassName = "",
    hoverGlow = false,
    hoverGlowStrength = "medium",
    onPointerDown,
    onPointerMove,
    style: styleProp,
    visualProps,
    ...elementProps
  }: ViComponentSurfaceProps<T>,
  ref: ComponentPropsWithRef<T>["ref"],
) {
  const Component = (as ?? "div") as ElementType;
  const resolvedConfig = config ?? createViVisualConfig(visualProps);

  function handlePointerMove(event: PointerEvent<HTMLElement>) {
    const rect = event.currentTarget.getBoundingClientRect();
    if (rect.width === 0 || rect.height === 0) {
      onPointerMove?.(event);
      return;
    }

    const x = Math.min(100, Math.max(0, ((event.clientX - rect.left) / rect.width) * 100));
    const y = Math.min(100, Math.max(0, ((event.clientY - rect.top) / rect.height) * 100));

    event.currentTarget.style.setProperty("--vi-glow-x", `${x.toFixed(2)}%`);
    event.currentTarget.style.setProperty("--vi-glow-y", `${y.toFixed(2)}%`);
    onPointerMove?.(event);
  }

  const style: ViSurfaceStyle = {
    ...createViVisualStyle(resolvedConfig),
    ...styleProp,
    "--vi-glow-x": "50%",
    "--vi-glow-y": "50%",
  };
  const Content = contentAs;

  return (
    <Component
      {...elementProps}
      ref={ref}
      className={`vi-component-surface ${className}`.trim()}
      data-focus-contract="vi-surface"
      data-glass={resolvedConfig.glass}
      data-hover-glow={hoverGlow}
      data-hover-glow-strength={hoverGlowStrength}
      data-shiny={resolvedConfig.shinyBorder}
      onPointerDown={onPointerDown}
      onPointerMove={hoverGlow ? handlePointerMove : onPointerMove}
      style={style}
    >
      <span className="vi-component-surface__shine" aria-hidden="true" />
      <span className="vi-component-surface__hover-glow" aria-hidden="true" />
      <Content className={`vi-component-surface__content ${contentClassName}`.trim()}>
        {children}
      </Content>
    </Component>
  );
}

const ViComponentSurfaceForwardRef = forwardRef(
  ViComponentSurfaceImpl as unknown as ForwardRefRenderFunction<unknown, ViComponentSurfaceProps>,
);

export const ViComponentSurface = ViComponentSurfaceForwardRef as <
  T extends ElementType = "div",
>(
  props: ViComponentSurfaceProps<T> & { ref?: ComponentPropsWithRef<T>["ref"] },
) => ReactElement | null;

export type { ViComponentSurfaceOwnProps };
