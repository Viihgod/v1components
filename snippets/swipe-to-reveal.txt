"use client";

import {
  forwardRef,
  useEffect,
  useRef,
  useState,
  type CSSProperties,
  type HTMLAttributes,
  type KeyboardEvent,
  type PointerEvent,
  type ReactNode,
} from "react";

import type { ViMotionProps } from "../../core/motion-props";
import {
  createViVisualConfig,
  createViVisualStyle,
  type ViVisualProps,
} from "../../core/visual-props";
import { useViReducedMotion } from "../shared/action-feedback-utils";
import {
  clamp,
  getPointerPoint,
  isInteractiveTarget,
  releasePointerCapture,
  thresholdPixels,
  useControllableValue,
  type ViDragStyle,
  type ViPointerPoint,
  type ViSwipeDirection,
} from "../shared/b05-drag-utils";
import "../shared/b05-drag.css";

export type ViSwipeAction = {
  disabled?: boolean;
  id: string;
  label: string;
  onSelect?: () => void;
};

export type ViSwipeToRevealProps = Omit<HTMLAttributes<HTMLDivElement>, "children"> &
  ViVisualProps &
  ViMotionProps & {
    actions: readonly ViSwipeAction[];
    children: ReactNode;
    defaultOpen?: boolean;
    disabled?: boolean;
    direction?: ViSwipeDirection;
    onOpenChange?: (open: boolean) => void;
    open?: boolean;
    threshold?: number;
  };

type SwipeStyle = ViDragStyle & { "--vi-swipe-open-distance": string };

export const ViSwipeToReveal = forwardRef<HTMLDivElement, ViSwipeToRevealProps>(
  function ViSwipeToReveal(
    {
      actions,
      activeColor,
      activeTextColor,
      borderColor,
      children,
      className = "",
      defaultOpen = false,
      direction = "left",
      disabled = false,
      glass,
      motion = "auto",
      motionConfig,
      onBlur,
      onKeyDown,
      onOpenChange,
      onPointerCancel,
      onPointerDown,
      onPointerMove,
      onPointerUp,
      open: openProp,
      paused,
      radius = "md",
      shineColorA,
      shineColorB,
      shinyBorder,
      standbyColor,
      standbyTextColor,
      style,
      threshold = 0.35,
      ...rest
    },
    ref,
  ) {
    const [open, setOpen, controlled] = useControllableValue(openProp, defaultOpen);
    const [dragging, setDragging] = useState(false);
    const pointerRef = useRef<(ViPointerPoint & { startOffset: number }) | null>(null);
    const pointerElementRef = useRef<HTMLElement | null>(null);
    const rootRef = useRef<HTMLDivElement | null>(null);
    const reducedMotion = useViReducedMotion(motion);
    const sign = direction === "left" ? -1 : 1;
    const openDistance = Math.max(96, actions.length * 80);
    const visual = createViVisualConfig({
      activeColor,
      activeTextColor,
      borderColor,
      glass,
      radius,
      shineColorA,
      shineColorB,
      shinyBorder,
      standbyColor,
      standbyTextColor,
    });
    const swipeStyle: SwipeStyle = {
      ...createViVisualStyle(visual),
      "--vi-drag-x": open ? "calc(min(100%, var(--vi-swipe-open-distance)) * " + sign + ")" : "0px",
      "--vi-swipe-open-distance": `${openDistance}px`,
      ...style,
    } as CSSProperties as SwipeStyle;
    void motionConfig;
    void paused;

    function setRootRef(node: HTMLDivElement | null) {
      rootRef.current = node;
      if (typeof ref === "function") ref(node);
      else if (ref) ref.current = node;
    }

    function restoreOffset(next: boolean) {
      rootRef.current?.style.setProperty("--vi-drag-x", next ? "calc(min(100%, var(--vi-swipe-open-distance)) * " + sign + ")" : "0px");
    }

    function setOpenState(next: boolean, restoreFocus = false) {
      if (disabled) return;
      if (!controlled) setOpen(next);
      onOpenChange?.(next);
      restoreOffset(next);
      if (restoreFocus) queueMicrotask(() => rootRef.current?.focus());
    }

    function cancelPointer(element: HTMLElement) {
      releasePointerCapture(element, pointerRef.current?.pointerId ?? null);
      pointerRef.current = null;
      pointerElementRef.current = null;
      setDragging(false);
      restoreOffset(open);
    }

    useEffect(() => () => {
      releasePointerCapture(pointerElementRef.current, pointerRef.current?.pointerId ?? null);
      pointerRef.current = null;
      pointerElementRef.current = null;
    }, []);

    function handleKeyDown(event: KeyboardEvent<HTMLDivElement>) {
      onKeyDown?.(event);
      if (event.defaultPrevented || disabled) return;
      const opens = direction === "left" ? event.key === "ArrowLeft" : event.key === "ArrowRight";
      const closes = direction === "left" ? event.key === "ArrowRight" : event.key === "ArrowLeft";
      if (opens || event.key === "Enter" || event.key === " ") {
        event.preventDefault();
        setOpenState(true);
      } else if (closes || event.key === "Escape") {
        event.preventDefault();
        setOpenState(false, true);
      }
    }

    return (
      <div
        {...rest}
        ref={setRootRef}
        className={`vi-b05-control vi-b05-swipe vi-b05-focus ${className.trim()}`}
        data-direction={direction}
        data-disabled={disabled}
        data-dragging={dragging}
        data-motion={reducedMotion ? "reduced" : "auto"}
        data-open={open}
        role="group"
        tabIndex={disabled ? -1 : (rest.tabIndex ?? 0)}
        aria-label={rest["aria-label"] ?? "Swipe actions"}
        style={swipeStyle}
        onBlur={(event) => {
          if (pointerRef.current) cancelPointer(event.currentTarget);
          onBlur?.(event);
        }}
        onKeyDown={handleKeyDown}
        onPointerCancel={(event) => {
          cancelPointer(event.currentTarget);
          onPointerCancel?.(event);
        }}
        onPointerDown={(event: PointerEvent<HTMLDivElement>) => {
          onPointerDown?.(event);
          if (
            event.defaultPrevented ||
            disabled ||
            isInteractiveTarget(event.target) ||
            (event.button !== 0 && event.button !== -1 && event.button !== undefined)
          ) return;
          pointerRef.current = { ...getPointerPoint(event), startOffset: open ? sign * Math.min(openDistance, event.currentTarget.clientWidth) : 0 };
          pointerElementRef.current = event.currentTarget;
          event.currentTarget.setPointerCapture?.(event.pointerId);
          setDragging(true);
        }}
        onPointerMove={(event) => {
          onPointerMove?.(event);
          const pointer = pointerRef.current;
          if (!pointer || pointer.pointerId !== event.pointerId) return;
          const travel = Math.min(openDistance, event.currentTarget.clientWidth);
          const distance = clamp(pointer.startOffset + (event.clientX - pointer.clientX), sign < 0 ? -travel : 0, sign < 0 ? 0 : travel);
          event.currentTarget.style.setProperty("--vi-drag-x", `${distance}px`);
        }}
        onPointerUp={(event) => {
          onPointerUp?.(event);
          const pointer = pointerRef.current;
          if (!pointer) return;
          const delta = event.clientX - pointer.clientX;
          const rect = event.currentTarget.getBoundingClientRect();
          const passed = sign * (pointer.startOffset + delta) >= thresholdPixels(threshold, rect.width);
          releasePointerCapture(event.currentTarget, pointer.pointerId);
          pointerRef.current = null;
          pointerElementRef.current = null;
          setDragging(false);
          if (passed) setOpenState(true);
          else setOpenState(false);
        }}
      >
        <div className="vi-b05-swipe__content">{children}</div>
        <div className="vi-b05-swipe__actions" role="group" aria-label="Swipe actions">
          {actions.map((action) => (
            <button
              key={action.id}
              type="button"
              className="vi-b05-swipe__action vi-b05-focus"
              disabled={disabled || action.disabled}
              tabIndex={open && !disabled && !action.disabled ? 0 : -1}
              onClick={() => {
                action.onSelect?.();
                setOpenState(false, true);
              }}
            >
              {action.label}
            </button>
          ))}
        </div>
      </div>
    );
  },
);

"use client";

import { useState, type CSSProperties, type ReactNode } from "react";

export type ViDragAxis = "horizontal" | "vertical";
export type ViSwipeDirection = "left" | "right";
export type ViLassoCollisionMode = "intersect" | "center";

export type ViPointerPoint = {
  clientX: number;
  clientY: number;
  pointerId: number;
};

export type ViDragStyle = CSSProperties & {
  "--vi-drag-x"?: string;
  "--vi-drag-y"?: string;
  "--vi-lasso-x"?: string;
  "--vi-lasso-y"?: string;
  "--vi-lasso-width"?: string;
  "--vi-lasso-height"?: string;
  "--vi-stack-index"?: number;
  "--vi-stack-fan"?: string;
};

export const MAX_LASSO_ITEMS = 128;

export function clamp(value: number, min: number, max: number) {
  if (!Number.isFinite(value)) return min;
  return Math.min(max, Math.max(min, value));
}

export function normalizeUnit(value: number | undefined, fallback: number) {
  return clamp(value ?? fallback, 0, 1);
}

export function useControllableValue<T>(controlled: T | undefined, defaultValue: T) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  const isControlled = controlled !== undefined;
  return [isControlled ? controlled : uncontrolled, setUncontrolled, isControlled] as const;
}

export function getPointerPoint(event: { clientX: number; clientY: number; pointerId: number }): ViPointerPoint {
  return { clientX: event.clientX, clientY: event.clientY, pointerId: event.pointerId };
}

export function getDelta(start: ViPointerPoint, current: ViPointerPoint) {
  return { x: current.clientX - start.clientX, y: current.clientY - start.clientY };
}

export function axisDelta(delta: { x: number; y: number }, axis: ViDragAxis) {
  return axis === "horizontal" ? delta.x : delta.y;
}

export function thresholdPixels(threshold: number | undefined, size: number, fallback = 0.35) {
  const safe = Number.isFinite(threshold) ? Math.abs(threshold as number) : fallback;
  return safe <= 1 ? Math.max(1, size * safe) : safe;
}

export function releasePointerCapture(element: HTMLElement | null, pointerId: number | null) {
  if (element && pointerId !== null && element.hasPointerCapture?.(pointerId)) {
    element.releasePointerCapture(pointerId);
  }
}

export function isInteractiveTarget(target: EventTarget | null) {
  return target instanceof Element && Boolean(
    target.closest("button,a,input,textarea,select,summary,[contenteditable='true'],[role='button'],[role='link']"),
  );
}

export function moveArray<T>(items: readonly T[], from: number, to: number) {
  if (from === to || from < 0 || to < 0 || from >= items.length || to >= items.length) return [...items];
  const next = [...items];
  const [item] = next.splice(from, 1);
  next.splice(to, 0, item);
  return next;
}

export function nextIndex(index: number, count: number, direction: -1 | 1, wrap = false) {
  if (!count) return -1;
  const next = index + direction;
  if (wrap) return (next + count) % count;
  return clamp(next, 0, count - 1);
}

export function itemId<T>(item: T, index: number, getId?: (item: T, index: number) => string) {
  if (getId) return getId(item, index);
  if (typeof item === "object" && item !== null && "id" in item) {
    const id = (item as { id?: unknown }).id;
    if (typeof id === "string" || typeof id === "number") return String(id);
  }
  return `item-${index}`;
}

export function rectContains(rect: DOMRect, x: number, y: number) {
  return x >= rect.left && x <= rect.right && y >= rect.top && y <= rect.bottom;
}

export function rectIntersects(a: DOMRect, b: DOMRect) {
  return a.left <= b.right && a.right >= b.left && a.top <= b.bottom && a.bottom >= b.top;
}

export function createAnnouncement(message: string) {
  return message;
}

export function renderMaybe<T>(render: ((item: T, index: number) => ReactNode) | undefined, item: T, index: number, fallback: ReactNode) {
  return render ? render(item, index) : fallback;
}

"use client";

import { useEffect, useRef, useState, useSyncExternalStore, type CSSProperties } from "react";

import type { ViMotionMode } from "../../core/motion-props";

/** The small style extension shared by action feedback components. */
export type ViActionStyle = CSSProperties & Record<`--vi-${string}`, string | number>;

export function clamp(value: number, min: number, max: number) {
  if (!Number.isFinite(value)) return min;
  return Math.min(max, Math.max(min, value));
}

export function isViReducedMotion(mode: ViMotionMode = "auto") {
  if (mode === "none") return true;
  return typeof window !== "undefined" &&
    typeof window.matchMedia === "function" &&
    Boolean(window.matchMedia("(prefers-reduced-motion: reduce)")?.matches);
}

function subscribeReducedMotion(onStoreChange: () => void) {
  if (typeof window === "undefined" || typeof window.matchMedia !== "function") {
    return () => {};
  }

  const media = window.matchMedia("(prefers-reduced-motion: reduce)");
  const update = () => onStoreChange();
  media.addEventListener?.("change", update);
  media.addListener?.(update);

  return () => {
    media.removeEventListener?.("change", update);
    media.removeListener?.(update);
  };
}

function getReducedMotionSnapshot() {
  return isViReducedMotion("auto");
}

function getReducedMotionServerSnapshot() {
  return false;
}

function noopSubscribe() {
  return () => {};
}

function alwaysReduced() {
  return true;
}

export function useViReducedMotion(mode: ViMotionMode = "auto") {
  const prefersReducedMotion = useSyncExternalStore(
    mode === "none" ? noopSubscribe : subscribeReducedMotion,
    mode === "none" ? alwaysReduced : getReducedMotionSnapshot,
    mode === "none" ? alwaysReduced : getReducedMotionServerSnapshot,
  );

  return mode === "none" || prefersReducedMotion;
}

export function useControllableBoolean(
  controlled: boolean | undefined,
  defaultValue: boolean,
) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  const value = controlled ?? uncontrolled;

  return [value, setUncontrolled] as const;
}

/** Tracks finite timers so transient feedback cannot outlive its component. */
export function useActionTimeout() {
  const timers = useRef(new Set<number>());

  useEffect(() => {
    const activeTimers = timers.current;
    return () => {
      for (const timer of activeTimers) window.clearTimeout(timer);
      activeTimers.clear();
    };
  }, []);

  return (callback: () => void, delayMs: number) => {
    const timer = window.setTimeout(() => {
      timers.current.delete(timer);
      callback();
    }, delayMs);
    timers.current.add(timer);
    return timer;
  };
}
