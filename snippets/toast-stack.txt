export type ViToastStackPlacement = ViAsyncPlacement;
export type ViToastStackTone = "info" | "success" | "error" | "warning";

export type ViToastStackItem = {
  actionLabel?: string;
  durationMs?: number;
  id: string;
  message: ReactNode;
  onAction?: () => void;
  tone?: ViToastStackTone;
};

export type ViToastStackProps = Omit<HTMLAttributes<HTMLDivElement>, "children"> &
  ViVisualProps &
  ViMotionProps & {
    durationMs?: number;
    items: readonly ViToastStackItem[];
    maxVisible?: number;
    onDismiss?: (id: string) => void;
    placement?: ViToastStackPlacement;
  };

function ToastStackEntry({
  item,
  durationMs,
  motionReduced,
  onDismiss,
  paused = false,
}: {
  item: ViToastStackItem;
  durationMs: number;
  motionReduced: boolean;
  onDismiss?: (id: string) => void;
  paused?: boolean;
}) {
  const rootRef = useRef<HTMLDivElement | null>(null);
  const announcement = announceableText(item.message, "Notification");
  const { handlers } = useViTransientTimer({
    active: true,
    delayMs: item.durationMs ?? durationMs,
    onExpire: () => onDismiss?.(item.id),
    paused,
  });
  useViRestoreFocus(true, rootRef);

  return (
    <div
      ref={rootRef}
      className="vi-b06-stack__item"
      data-kind={item.tone ?? "info"}
      data-motion={motionReduced ? "reduced" : "auto"}
      data-open="true"
      onBlur={handlers.onBlur}
      onFocus={handlers.onFocus}
      onPointerEnter={handlers.onPointerEnter}
      onPointerLeave={handlers.onPointerLeave}
      role="status"
      aria-live="off"
      aria-label={announcement}
    >
      <span className="vi-b06-stack__message">{item.message}</span>
      {item.actionLabel ? (
        <button
          type="button"
          className="vi-b06-transient__action vi-b06-focus"
          onClick={() => {
            item.onAction?.();
            onDismiss?.(item.id);
          }}
        >
          {item.actionLabel}
        </button>
      ) : null}
      <button
        type="button"
        className="vi-b06-transient__dismiss vi-b06-stack__close vi-b06-focus"
        aria-label={`Dismiss ${announceableText(item.message, "notification")}`}
        onClick={() => onDismiss?.(item.id)}
      >
        ×
      </button>
      <ViAsyncLiveRegion
        active
        announcement={announcement}
        id={`vi-b06-stack-live-${item.id}`}
      />
    </div>
  );
}

export const ViToastStack = forwardRef<HTMLDivElement, ViToastStackProps>(
  function ViToastStack(
    {
      activeColor,
      activeTextColor,
      borderColor,
      className = "",
      durationMs = 5000,
      glass,
      items,
      maxVisible = 3,
      motion = "auto",
      motionConfig,
      onDismiss,
      paused,
      placement = "top-end",
      radius = "md",
      shineColorA,
      shineColorB,
      shinyBorder,
      standbyColor,
      standbyTextColor,
      style,
      ...rest
    },
    ref,
  ) {
    const reducedMotion = useViReducedMotion(motion);
    const visual = createViVisualConfig({
      activeColor,
      activeTextColor,
      borderColor,
      glass,
      radius,
      shineColorA,
      shineColorB,
      shinyBorder,
      standbyColor,
      standbyTextColor,
    });
    const visibleItems = items.slice(0, visibleToastCount(maxVisible, items.length));
    void motionConfig;
    void paused;

    return (
      <div
        {...rest}
        ref={ref}
        className={`vi-b06-control vi-b06-stack ${className.trim()}`.trim()}
        data-motion={reducedMotion ? "reduced" : "auto"}
        data-placement={placement}
        aria-label={rest["aria-label"] ?? "Notification stack"}
        role="region"
        style={{ ...createViVisualStyle(visual), ...style }}
      >
        {visibleItems.map((item) => (
          <ToastStackEntry
            key={item.id}
            item={item}
            durationMs={durationMs}
            motionReduced={reducedMotion}
            onDismiss={onDismiss}
            paused={paused}
          />
        ))}
        {visibleItems.length === 0 ? <span className="vi-b06-composer__status">No notifications</span> : null}
      </div>
    );
  },
);

"use client";

import {
  forwardRef,
  useCallback,
  useEffect,
  useId,
  useRef,
  useState,
  type FocusEvent,
  type CSSProperties,
  type HTMLAttributes,
  type ReactNode,
} from "react";

import { clamp } from "./action-feedback-utils";
export { useViReducedMotion } from "./action-feedback-utils";

export type ViAsyncPlacement =
  | "top-start"
  | "top"
  | "top-end"
  | "bottom-start"
  | "bottom"
  | "bottom-end";

export type ViAsyncStatus = "idle" | "pending" | "success" | "error" | "warning" | "info";

export type ViTransientHandlers = {
  onBlur: (event: FocusEvent<HTMLElement>) => void;
  onFocus: () => void;
  onPointerEnter: () => void;
  onPointerLeave: () => void;
};

export function useControllableAsyncBoolean(
  controlled: boolean | undefined,
  defaultValue: boolean,
) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  const isControlled = controlled !== undefined;
  return [isControlled ? controlled : uncontrolled, setUncontrolled, isControlled] as const;
}

function useDocumentHidden() {
  const [hidden, setHidden] = useState(false);

  useEffect(() => {
    const update = () => setHidden(document.hidden);
    update();
    document.addEventListener("visibilitychange", update);
    return () => document.removeEventListener("visibilitychange", update);
  }, []);

  return hidden;
}

/**
 * Schedules one finite timeout while preserving its remaining duration when
 * hover, focus, document visibility or an explicit paused prop interrupts it.
 */
export function useViTransientTimer({
  active,
  delayMs,
  onExpire,
  paused = false,
}: {
  active: boolean;
  delayMs: number;
  onExpire: () => void;
  paused?: boolean;
}) {
  const [hovered, setHovered] = useState(false);
  const [focused, setFocused] = useState(false);
  const documentHidden = useDocumentHidden();
  const delay = Math.max(0, Number.isFinite(delayMs) ? delayMs : 0);
  const remainingRef = useRef(delay);
  const timeoutRef = useRef<number | null>(null);
  const startedAtRef = useRef<number | null>(null);
  const activeRef = useRef(false);
  const onExpireRef = useRef(onExpire);

  useEffect(() => {
    onExpireRef.current = onExpire;
  }, [onExpire]);

  const clearTimer = useCallback(() => {
    if (timeoutRef.current !== null) {
      window.clearTimeout(timeoutRef.current);
      timeoutRef.current = null;
    }
    if (startedAtRef.current !== null) {
      remainingRef.current = Math.max(
        0,
        remainingRef.current - (Date.now() - startedAtRef.current),
      );
      startedAtRef.current = null;
    }
  }, []);

  const isPaused = paused || hovered || focused || documentHidden;

  useEffect(() => {
    clearTimer();

    if (!active) {
      activeRef.current = false;
      remainingRef.current = delay;
      return;
    }

    if (!activeRef.current) {
      remainingRef.current = delay;
      activeRef.current = true;
    }

    if (isPaused || delay <= 0) return;

    const wait = Math.max(1, remainingRef.current);
    startedAtRef.current = Date.now();
    timeoutRef.current = window.setTimeout(() => {
      timeoutRef.current = null;
      startedAtRef.current = null;
      remainingRef.current = delay;
      onExpireRef.current();
    }, wait);

    return clearTimer;
  }, [active, clearTimer, delay, isPaused]);

  useEffect(() => () => clearTimer(), [clearTimer]);

  const onBlur = useCallback((event: FocusEvent<HTMLElement>) => {
    const next = event.relatedTarget;
    if (!next || !event.currentTarget.contains(next as Node)) setFocused(false);
  }, []);

  return {
    handlers: {
      onBlur,
      onFocus: () => setFocused(true),
      onPointerEnter: () => setHovered(true),
      onPointerLeave: () => setHovered(false),
    } satisfies ViTransientHandlers,
    paused: isPaused,
  };
}

export function useViRestoreFocus(active: boolean, rootRef: { current: HTMLElement | null }) {
  const previousFocusRef = useRef<HTMLElement | null>(null);
  const wasActiveRef = useRef(false);

  const restore = useCallback(() => {
    const target = previousFocusRef.current;
    const root = rootRef.current;
    if (target && root?.contains(document.activeElement)) {
      target.focus({ preventScroll: true });
    }
    previousFocusRef.current = null;
  }, [rootRef]);

  useEffect(() => {
    if (active && !wasActiveRef.current) {
      const candidate = document.activeElement;
      if (candidate instanceof HTMLElement && !rootRef.current?.contains(candidate)) {
        previousFocusRef.current = candidate;
      }
    } else if (!active && wasActiveRef.current) {
      restore();
    }
    wasActiveRef.current = active;
  }, [active, restore, rootRef]);

  useEffect(() => () => restore(), [restore]);
}

export function ViAsyncLiveRegion({
  active,
  announcement,
  assertive = false,
  id,
}: {
  active: boolean;
  announcement: string;
  assertive?: boolean;
  id: string;
}) {
  const key = `${id}-${active ? announcement : ""}`;
  return (
    <span
      key={key}
      id={id}
      className="vi-b06-live"
      aria-atomic="true"
      aria-live={assertive ? "assertive" : "polite"}
    >
      {active ? announcement : ""}
    </span>
  );
}

export function announceableText(value: ReactNode, fallback: string) {
  if (typeof value === "string" || typeof value === "number") return String(value);
  return fallback;
}

export function normalizeAsyncStatus(status: string): ViAsyncStatus {
  if (
    status === "idle" ||
    status === "pending" ||
    status === "success" ||
    status === "error" ||
    status === "warning" ||
    status === "info"
  ) return status;
  return "info";
}

export function visibleToastCount(value: number | undefined, length: number) {
  const max = Math.floor(Number.isFinite(value) ? Number(value) : 3);
  return Math.min(length, Math.max(1, max));
}

export function clampAsyncRows(value: number | undefined, fallback: number) {
  return clamp(Math.round(Number.isFinite(value) ? Number(value) : fallback), 1, 12);
}

export type ViAsyncTransientKind = "toast" | "notification" | "undo";

export type ViAsyncTransientProps = Omit<HTMLAttributes<HTMLDivElement>, "children"> & {
  actionLabel?: string;
  active: boolean;
  announcement: string;
  children: ReactNode;
  kind: ViAsyncTransientKind;
  onAction?: () => void;
  onRequestClose: (reason: "action" | "escape" | "timeout" | "dismiss") => void;
  paused?: boolean;
  placement: ViAsyncPlacement;
  durationMs: number;
  motionReduced: boolean;
  style?: CSSProperties;
};

export const ViAsyncTransient = forwardRef<HTMLDivElement, ViAsyncTransientProps>(
  function ViAsyncTransient(
    {
      actionLabel,
      active,
      announcement,
      children,
      className = "",
      durationMs,
      id,
      kind,
      motionReduced,
      onAction,
      onBlur: onNativeBlur,
      onFocus: onNativeFocus,
      onKeyDown: onNativeKeyDown,
      onPointerEnter: onNativePointerEnter,
      onPointerLeave: onNativePointerLeave,
      onRequestClose,
      paused = false,
      placement,
      role: nativeRole,
      style,
      ...rest
    },
    forwardedRef,
  ) {
    const rootRef = useRef<HTMLDivElement | null>(null);
    const generatedId = useId().replaceAll(":", "");
    const liveId = id ? `${id}-live` : `vi-b06-live-${generatedId}`;
    const { handlers } = useViTransientTimer({
      active,
      delayMs: durationMs,
      onExpire: () => onRequestClose("timeout"),
      paused,
    });
    useViRestoreFocus(active, rootRef);

    function setRootRef(node: HTMLDivElement | null) {
      rootRef.current = node;
      if (typeof forwardedRef === "function") forwardedRef(node);
      else if (forwardedRef) forwardedRef.current = node;
    }

    function handleKeyDown(event: React.KeyboardEvent<HTMLDivElement>) {
      onNativeKeyDown?.(event);
      if (event.defaultPrevented || event.key !== "Escape" || !active) return;
      event.preventDefault();
      onRequestClose("escape");
    }

    function handleAction() {
      onAction?.();
      onRequestClose("action");
    }

    return (
      <div
        {...rest}
        ref={setRootRef}
        id={id}
        className={`vi-b06-control vi-b06-transient-host ${className.trim()}`.trim()}
        data-motion={motionReduced ? "reduced" : "auto"}
        data-open={active}
        onBlur={(event) => {
          handlers.onBlur(event);
          onNativeBlur?.(event);
        }}
        onFocus={(event) => {
          handlers.onFocus();
          onNativeFocus?.(event);
        }}
        onKeyDown={handleKeyDown}
        onPointerEnter={(event) => {
          handlers.onPointerEnter();
          onNativePointerEnter?.(event);
        }}
        onPointerLeave={(event) => {
          handlers.onPointerLeave();
          onNativePointerLeave?.(event);
        }}
        role={nativeRole ?? "status"}
        aria-hidden={!active}
        style={style}
      >
        <div
          className={`vi-b06-transient vi-b06-transient--${kind}`}
          data-placement={placement}
          aria-hidden={!active}
        >
          <span className="vi-b06-transient__message">{children}</span>
          {actionLabel ? (
            <button
              type="button"
              className="vi-b06-transient__action vi-b06-focus"
              tabIndex={active ? 0 : -1}
              onClick={handleAction}
              disabled={!active}
            >
              {actionLabel}
            </button>
          ) : null}
          <button
            type="button"
            className="vi-b06-transient__dismiss vi-b06-focus"
            aria-label="Dismiss notification"
            tabIndex={active ? 0 : -1}
            onClick={() => onRequestClose("dismiss")}
            disabled={!active}
          >
            ×
          </button>
        </div>
        <ViAsyncLiveRegion active={active} announcement={announcement} id={liveId} />
      </div>
    );
  },
);

// --- src/components/vicomponents/components/shared/action-feedback-utils.ts ---
"use client";

import { useEffect, useRef, useState, useSyncExternalStore, type CSSProperties } from "react";

import type { ViMotionMode } from "../../core/motion-props";

/** The small style extension shared by action feedback components. */
export type ViActionStyle = CSSProperties & Record<`--vi-${string}`, string | number>;

export function clamp(value: number, min: number, max: number) {
  if (!Number.isFinite(value)) return min;
  return Math.min(max, Math.max(min, value));
}

export function isViReducedMotion(mode: ViMotionMode = "auto") {
  if (mode === "none") return true;
  return typeof window !== "undefined" &&
    typeof window.matchMedia === "function" &&
    Boolean(window.matchMedia("(prefers-reduced-motion: reduce)")?.matches);
}

function subscribeReducedMotion(onStoreChange: () => void) {
  if (typeof window === "undefined" || typeof window.matchMedia !== "function") {
    return () => {};
  }

  const media = window.matchMedia("(prefers-reduced-motion: reduce)");
  const update = () => onStoreChange();
  media.addEventListener?.("change", update);
  media.addListener?.(update);

  return () => {
    media.removeEventListener?.("change", update);
    media.removeListener?.(update);
  };
}

function getReducedMotionSnapshot() {
  return isViReducedMotion("auto");
}

function getReducedMotionServerSnapshot() {
  return false;
}

function noopSubscribe() {
  return () => {};
}

function alwaysReduced() {
  return true;
}

export function useViReducedMotion(mode: ViMotionMode = "auto") {
  const prefersReducedMotion = useSyncExternalStore(
    mode === "none" ? noopSubscribe : subscribeReducedMotion,
    mode === "none" ? alwaysReduced : getReducedMotionSnapshot,
    mode === "none" ? alwaysReduced : getReducedMotionServerSnapshot,
  );

  return mode === "none" || prefersReducedMotion;
}

export function useControllableBoolean(
  controlled: boolean | undefined,
  defaultValue: boolean,
) {
  const [uncontrolled, setUncontrolled] = useState(defaultValue);
  const value = controlled ?? uncontrolled;

  return [value, setUncontrolled] as const;
}

/** Tracks finite timers so transient feedback cannot outlive its component. */
export function useActionTimeout() {
  const timers = useRef(new Set<number>());

  useEffect(() => {
    const activeTimers = timers.current;
    return () => {
      for (const timer of activeTimers) window.clearTimeout(timer);
      activeTimers.clear();
    };
  }, []);

  return (callback: () => void, delayMs: number) => {
    const timer = window.setTimeout(() => {
      timers.current.delete(timer);
      callback();
    }, delayMs);
    timers.current.add(timer);
    return timer;
  };
}

  },
) {
  return visual;
}
